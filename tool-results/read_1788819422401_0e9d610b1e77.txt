    16→// Tokenizer + parser s-expression (subset KiCad)

    18→
    19→// sNode is one node of the s-expression tree. A list node keeps its direct
    20→// atoms (the first one being the tag) and its direct sub-lists; a bare atom
    21→// is not represented outside of the atoms slice of its parent.
    22→type sNode struct {
    23→        atoms    []string
    24→        children []*sNode
    25→}
    26→
    27→// tag returns the tag of a list node ("segment", "pad", ...) or "".
    28→func (n *sNode) tag() string {
    29→        if n == nil || len(n.atoms) == 0 {
    30→                return ""
    31→        }
    32→        return n.atoms[0]
    33→}
    34→
    35→// args returns the atoms following the tag.
    36→func (n *sNode) args() []string {
    37→        if n == nil || len(n.atoms) <= 1 {
    38→                return nil
    39→        }
    40→        return n.atoms[1:]
    41→}
    42→
    43→// arg returns the i-th atom after the tag, or "" when missing.
    44→func (n *sNode) arg(i int) string {
    45→        args := n.args()
    46→        if i < 0 || i >= len(args) {
    47→                return ""
    48→        }
    49→        return args[i]
    50→}
    51→
    52→// child returns the first direct sub-list carrying the given tag.
    53→func (n *sNode) child(name string) *sNode {
    54→        for _, c := range n.children {
    55→                if c.tag() == name {
    56→                        return c
    57→                }
    58→        }
    59→        return nil
    60→}
    61→
    62→// childrenWithTag returns every direct sub-list carrying the given tag.
    63→func (n *sNode) childrenWithTag(name string) []*sNode {
    64→        var out []*sNode
    65→        for _, c := range n.children {
    66→                if c.tag() == name {
    67→                        out = append(out, c)
    68→                }
    69→        }
    70→        return out
    71→}
    72→
    73→// hasAtomIn reports whether one of the direct atoms after the tag equals v.
    74→func (n *sNode) hasAtomIn(v string) bool {
    75→        for _, a := range n.args() {
    76→                if a == v {
    77→                        return true
    78→                }
    79→        }
    80→        return false
    81→}

    83→// tokenizeSexpr splits an s-expression payload into "(", ")" and atom
    84→// tokens (quoted strings are unescaped).
    85→func tokenizeSexpr(data []byte) ([]string, error) {
    86→        var toks []string
    87→        i := 0
    88→        for i < len(data) {
    89→                c := data[i]
    90→                switch {
    91→                case c == '(' || c == ')':
    92→                        toks = append(toks, string(c))
    93→                        i++
    94→                case c == '"':
    95→                        s, next, err := readQuoted(data, i)
    96→                        if err != nil {
    97→                                return nil, err
    98→                        }
    99→                        toks = append(toks, s)
   100→                        i = next
   101→                case c == ' ' || c == '\t' || c == '\n' || c == '\r':
   102→                        i++
   103→                default:
   104→                        j := i
   105→                        for j < len(data) && data[j] != '(' && data[j] != ')' &&
   106→                                data[j] != '"' && data[j] != ' ' && data[j] != '\t' &&
   107→                                data[j] != '\n' && data[j] != '\r' {
   108→                                j++
   109→                        }
   110→                        toks = append(toks, string(data[i:j]))
   111→                        i = j
   112→                }
   113→        }
   114→        return toks, nil
   115→}
   116→
   117→// readQuoted reads a double-quoted atom starting at position start,
   118→// resolving the standard backslash escapes.
   119→func readQuoted(data []byte, start int) (string, int, error) {
   120→        var sb strings.Builder
   121→        i := start + 1
   122→        for i < len(data) {
   123→                c := data[i]
   124→                switch c {
   125→                case '"':
   126→                        return sb.String(), i + 1, nil
   127→                case '\\':
   128→                        if i+1 >= len(data) {
   129→                                return "", 0, fmt.Errorf("s-expression : échappement tronqué")
   130→                        }
   131→                        switch data[i+1] {
   132→                        case 'n':
   133→                                sb.WriteByte('\n')
   134→                        case 't':
   135→                                sb.WriteByte('\t')
   136→                        case 'r':
   137→                                sb.WriteByte('\r')
   138→                        default:
   139→                                sb.WriteByte(data[i+1])
   140→                        }
   141→                        i += 2
   142→                default:
   143→                        sb.WriteByte(c)
   144→                        i++
   145→                }
   146→        }
   147→        return "", 0, fmt.Errorf("s-expression : chaîne non terminée")
   148→}

   150→// parseSexpr parses a payload into a virtual root node holding every
   151→// top-level list (and every loose atom) as children.
   152→func parseSexpr(data []byte) (*sNode, error) {
   153→        toks, err := tokenizeSexpr(data)
   154→        if err != nil {
   155→                return nil, err
   156→        }
   157→        root := &sNode{}
   158→        pos := 0
   159→        for pos < len(toks) {
   160→                node, next, err := parseNode(toks, pos)
   161→                if err != nil {
   162→                        return nil, err
   163→                }
   164→                if node != nil {
   165→                        root.children = append(root.children, node)
   166→                }
   167→                pos = next
   168→        }
   169→        if len(root.children) == 0 {
   170→                return nil, fmt.Errorf("s-expression : fichier vide")
   171→        }
   172→        return root, nil
   173→}
   174→
   175→// parseNode parses one node starting at pos and returns the next position.
   176→func parseNode(toks []string, pos int) (*sNode, int, error) {
   177→        if pos >= len(toks) {
   178→                return nil, pos, fmt.Errorf("s-expression : fin inattendue")
   179→        }
   180→        if toks[pos] == ")" {
   181→                return nil, pos + 1, fmt.Errorf("s-expression : parenthèse fermante inattendue")
   182→        }
   183→        if toks[pos] != "(" {
   184→                // Atome isolé : représenté comme une liste dégénérée d'un seul atome,
   185→                // ignorée par les lecteurs (les atomes sont portés par leur parent).
   186→                return &sNode{atoms: []string{toks[pos]}}, pos + 1, nil
   187→        }
   188→        pos++
   189→        node := &sNode{}
   190→        for pos < len(toks) {
   191→                switch toks[pos] {
   192→                case ")":
   193→                        return node, pos + 1, nil
   194→                case "(":
   195→                        child, next, err := parseNode(toks, pos)
   196→                        if err != nil {
   197→                                return nil, pos, err
   198→                        }
   199→                        node.children = append(node.children, child)
   200→                        pos = next
   201→                default:
   202→                        node.atoms = append(node.atoms, toks[pos])
   203→                        pos++
   204→                }
   205→        }
   206→        return nil, pos, fmt.Errorf("s-expression : parenthèses déséquilibrées")
   207→}

   209→// sexprFloat parses a numeric atom with a zero fallback.
   210→func sexprFloat(s string) float64 {
   211→        v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
   212→        if err != nil || math.IsNaN(v) || math.IsInf(v, 0) {
   213→                return 0
   214→        }
   215→        return v
   216→}
   217→
   218→// sexprInt parses an integer atom with a zero fallback.
   219→func sexprInt(s string) int {
   220→        v, err := strconv.Atoi(strings.TrimSpace(s))
   221→        if err != nil {
   222→                return 0
   223→        }
   224→        return v
   225→}

   231→// kicadLayerIndex maps a KiCad layer name onto a copper layer index
   232→// (0 = F.Cu, layerCount-1 = B.Cu, InN.Cu = N).
   233→func kicadLayerIndex(name string, layerCount int) int {
   234→        switch {
   235→        case name == "F.Cu" || name == "Top":
   236→                return 0
   237→        case name == "B.Cu" || name == "Bottom":
   238→                if layerCount <= 1 {
   239→                        return 0
   240→                }
   241→                return layerCount - 1
   242→        case strings.HasPrefix(name, "In") && strings.HasSuffix(name, ".Cu"):
   243→                if n, err := strconv.Atoi(strings.TrimSuffix(strings.TrimPrefix(name, "In"), ".Cu")); err == nil {
   244→                        if n > 0 && n < layerCount-1 {
   245→                                return n
   246→                        }
   247→                }
   248→                return 0
   249→        default:
   250→                return 0
   251→        }
   252→}
   253→
   254→// padShapeFromKicad maps a KiCad pad shape onto the domain enum.
   255→func padShapeFromKicad(shape string) string {
   256→        switch strings.ToLower(strings.TrimSpace(shape)) {
   257→        case "circle":
   258→                return "circle"
   259→        case "oval", "ellipse":
   260→                return "oval"
   261→        case "roundrect", "round_rect":
   262→                return "roundrect"
   263→        case "rect":
   264→                return "rect"
   265→        default:
   266→                return "rect"
   267→        }
   268→}
   269→
   270→// stripLibraryPrefix removes the "Lib:" prefix of a footprint name.
   271→func stripLibraryPrefix(name string) string {
   272→        if idx := strings.LastIndex(name, ":"); idx >= 0 && idx < len(name)-1 {
   273→                return name[idx+1:]
   274→        }
   275→        return name
   276→}

   302→// geometryAABB is a minimal bounding box used for the board outline.
   303→type geometryAABB struct {
   304→        MinX, MinY, MaxX, MaxY float64
   305→}
   306→
   307→func (b *geometryAABB) unionPoint(x, y float64) {
   308→        b.MinX = math.Min(b.MinX, x)
   309→        b.MinY = math.Min(b.MinY, y)
   310→        b.MaxX = math.Max(b.MaxX, x)
   311→        b.MaxY = math.Max(b.MaxY, y)
   312→}
   313→
   314→func (b *geometryAABB) union(o geometryAABB) {
   315→        b.MinX = math.Min(b.MinX, o.MinX)
   316→        b.MinY = math.Min(b.MinY, o.MinY)
   317→        b.MaxX = math.Max(b.MaxX, o.MaxX)
   318→        b.MaxY = math.Max(b.MaxY, o.MaxY)
   319→}

   331→// readKiCadPCB parses a .kicad_pcb file (contract §7 subset) into the
   332→// schematic + board pair.
   333→func readKiCadPCB(data []byte, path string) (schematicapp.ImportResult, error) {
   334→        root, err := parseSexpr(data)
   335→        if err != nil {
   336→                return schematicapp.ImportResult{}, fmt.Errorf("kicad %s : %w", path, err)
   337→        }
   338→        pcb := root.child("kicad_pcb")
   339→        if pcb == nil {
   340→                return schematicapp.ImportResult{}, fmt.Errorf(
   341→                        "kicad %s : racine (kicad_pcb ...) introuvable", path)
   342→        }
   343→
   344→        st := &kicadParseState{
   345→                layerCount: 2,
   346→                netNames:   map[int]string{},
   347→                netUsed:    map[int]bool{},
   348→                conns:      map[int][]schematicPinRef{},
   349→        }
   350→
   351→        // Nombre de couches cuivre : entrées "xxx.Cu" du bloc (layers ...).
   352→        if layers := pcb.child("layers"); layers != nil {
   353→                cu := 0
   354→                for _, l := range layers.children {
   355→                        name := l.arg(0)
   356→                        if strings.HasSuffix(name, ".Cu") {
   357→                                cu++
   358→                        }
   359→                }
   360→                if cu >= 1 && cu <= 34 {
   361→                        st.layerCount = cu
   362→                }
   363→        }
   364→
   365→        // Déclarations (net N "NAME").
   366→        for _, n := range pcb.childrenWithTag("net") {
   367→                args := n.args()
   368→                if len(args) == 0 {
   369→                        continue
   370→                }
   371→                num := sexprInt(args[0])
   372→                name := ""
   373→                if len(args) > 1 {
   374→                        name = args[1]
   375→                }
   376→                if num > 0 && name != "" {
   377→                        if _, seen := st.netNames[num]; !seen {
   378→                                st.netOrder = append(st.netOrder, num)
   379→                        }
   380→                        st.netNames[num] = name
   381→                }
   382→        }
   383→
   384→        for _, fp := range pcb.childrenWithTag("footprint") {
   385→                st.readFootprint(fp)
   386→        }
   387→        for _, seg := range pcb.childrenWithTag("segment") {
   388→                st.readSegment(seg)
   389→        }
   390→        for _, via := range pcb.childrenWithTag("via") {
   391→                st.readVia(via)
   392→        }
   393→
   394→        width, height := st.boardOutline(pcb)
   395→        board, err := domainlayout.NewBoard(width, height, st.layerCount)
   396→        if err != nil {
   397→                return schematicapp.ImportResult{}, fmt.Errorf("kicad %s : %w", path, err)
   398→        }
   399→
   400→        // Empreintes -> composants placés.
   401→        for i := range st.footprints {
   402→                kf := &st.footprints[i]
   403→                bodyW, bodyH := kicadBodySize(kf.pads)
   404→                pl := domainlayout.PlacedComponent{
   405→                        Ref: kf.ref,
   406→                        Footprint: domainlayout.Footprint{
   407→                                Name:         kf.name,
   408→                                Value:        kf.value,
   409→                                Pads:         kf.pads,
   410→                                BodyWidthMM:  bodyW,
   411→                                BodyHeightMM: bodyH,
   412→                                HeightMM:     domainlayout.DefaultBoardHeightMM,
   413→                        },
   414→                        X:        kf.x,
   415→                        Y:        kf.y,
   416→                        Rotation: kf.rot,
   417→                }
   418→                if err := board.AddComponent(pl); err != nil {
   419→                        st.warnings = append(st.warnings, fmt.Sprintf("empreinte ignorée : %v", err))
   420→                        continue
   421→                }
   422→        }
   423→
   424→        // Pistes et vias.
   425→        for _, t := range st.tracks {
   426→                board.AddTrack(t)
   427→        }
   428→        for _, v := range st.vias {
   429→                if err := board.AddVia(v); err != nil {
   430→                        st.warnings = append(st.warnings, fmt.Sprintf("via ignoré : %v", err))
   431→                }
   432→        }
   433→
   434→        // Schéma : composants + nets reconstitués depuis les pastilles.
   435→        sch := domainschematic.New(schematicBaseName(path, "kicad"))
   436→        for i := range st.footprints {
   437→                kf := &st.footprints[i]
   438→                if err := sch.AddComponent(domainschematic.Component{
   439→                        Ref:       kf.ref,
   440→                        Value:     kf.value,
   441→                        Footprint: kf.name,
   442→                        X:         kf.x,
   443→                        Y:         kf.y,
   444→                        Rotation:  kf.rot,
   445→                }); err != nil {
   446→                        st.warnings = append(st.warnings, fmt.Sprintf("composant ignoré : %v", err))
   447→                }
   448→        }
   449→        for _, num := range st.netOrder {
   450→                if !st.netUsed[num] {
   451→                        continue
   452→                }
   453→                name := st.netNames[num]
   454→                if name == "" {
   455→                        name = fmt.Sprintf("NET_%d", num)
   456→                }
   457→                net := domainschematic.Net{Name: name, Class: domainschematic.ClassDefault}
   458→                for _, conn := range st.conns[num] {
   459→                        net.Connections = append(net.Connections, domainschematic.PinRef{
   460→                                ComponentRef: conn.ref, PinNumber: conn.pin,
   461→                        })
   462→                }
   463→                if err := sch.AddNet(net); err != nil {
   464→                        st.warnings = append(st.warnings, fmt.Sprintf("net ignoré : %v", err))
   465→                }
   466→        }
   467→
   468→        res := schematicapp.ImportResult{
   469→                Format:      "kicad",
   470→                Schematic:   sch,
   471→                Board:       board,
   472→                Constraints: domainconstraints.NewDefault(),
   473→                Warnings:    st.warnings,
   474→        }
   475→        return res, nil
   476→}

   478→// readFootprint parses one (footprint ...) node.
   479→func (st *kicadParseState) readFootprint(fp *sNode) {
   480→        kf := kicadFootprint{
   481→                name:  stripLibraryPrefix(fp.arg(0)),
   482→                front: true,
   483→        }
   484→
   485→        if layer := fp.child("layer"); layer != nil {
   486→                kf.front = layer.arg(0) != "B.Cu"
   487→        }
   488→        if at := fp.child("at"); at != nil {
   489→                kf.x = sexprFloat(at.arg(0))
   490→                kf.y = sexprFloat(at.arg(1))
   491→                if at.arg(2) != "" {
   492→                        kf.rot = sexprFloat(at.arg(2))
   493→                }
   494→        }
   495→
   496→        ref := fmt.Sprintf("FP%d", len(st.footprints)+1)
   497→        value := ""
   498→        // pcbnew ≥ 7 stocke référence et valeur dans des (property "Reference" /
   499→        // "Value" …) ; pcbnew ≤ 6 utilisait (fp_text reference/value …).
   500→        for _, prop := range fp.childrenWithTag("property") {
   501→                switch prop.arg(0) {
   502→                case "Reference":
   503→                        if v := prop.arg(1); v != "" {
   504→                                ref = v
   505→                        }
   506→                case "Value":
   507→                        if v := prop.arg(1); v != "" {
   508→                                value = v
   509→                        }
   510→                }
   511→        }
   512→        for _, txt := range fp.childrenWithTag("fp_text") {
   513→                switch txt.arg(0) {
   514→                case "reference":
   515→                        if v := txt.arg(1); v != "" {
   516→                                ref = v
   517→                        }
   518→                case "value":
   519→                        value = txt.arg(1)
   520→                }
   521→        }
   522→        kf.ref, kf.value = ref, value
   523→
   524→        lastLayer := st.layerCount - 1
   525→        for _, padNode := range fp.childrenWithTag("pad") {
   526→                args := padNode.args()
   527→                padName := ""
   528→                padType := "smd"
   529→                padShape := "rect"
   530→                if len(args) > 0 {
   531→                        padName = args[0]
   532→                }
   533→                if len(args) > 1 {
   534→                        padType = args[1]
   535→                }
   536→                if len(args) > 2 {
   537→                        padShape = args[2]
   538→                }
   539→
   540→                pad := domainlayout.Pad{
   541→                        Name:  padName,
   542→                        Shape: domainlayout.PadShape(padShapeFromKicad(padShape)),
   543→                }
   544→                if at := padNode.child("at"); at != nil {
   545→                        pad.X = sexprFloat(at.arg(0))
   546→                        pad.Y = sexprFloat(at.arg(1))
   547→                        pad.Rotation = sexprFloat(at.arg(2))
   548→                }
   549→                if size := padNode.child("size"); size != nil {
   550→                        pad.Width = sexprFloat(size.arg(0))
   551→                        pad.Height = sexprFloat(size.arg(1))
   552→                }
   553→                if drill := padNode.child("drill"); drill != nil {
   554→                        // (drill 1.0) ou (drill oval 1.0 0.6)
   555→                        d := sexprFloat(drill.arg(0))
   556→                        if d == 0 {
   557→                                d = sexprFloat(drill.arg(1))
   558→                        }
   559→                        _ = d // le perçage est porté par le diamètre côté domaine
   560→                }
   561→
   562→                layersNode := padNode.child("layers")
   563→                thru := padType == "thru_hole" || padType == "np_thru_hole"
   564→                if layersNode != nil && layersNode.hasAtomIn("*.Cu") {
   565→                        thru = true
   566→                }
   567→                switch {
   568→                case thru:
   569→                        pad.Layer = -1
   570→                case !kf.front:
   571→                        pad.Layer = lastLayer
   572→                default:
   573→                        pad.Layer = 0
   574→                }
   575→
   576→                // (net N "NAME") : association électrique de la pastille.
   577→                if netNode := padNode.child("net"); netNode != nil {
   578→                        args := netNode.args()
   579→                        if len(args) > 0 {
   580→                                num := sexprInt(args[0])
   581→                                name := ""
   582→                                if len(args) > 1 {
   583→                                        name = args[1]
   584→                                }
   585→                                if num > 0 && (name != "" || st.netNames[num] != "") {
   586→                                        if name != "" {
   587→                                                if _, seen := st.netNames[num]; !seen {
   588→                                                        st.netOrder = append(st.netOrder, num)
   589→                                                }
   590→                                                st.netNames[num] = name
   591→                                        }
   592→                                        pad.Net = st.netName(num)
   593→                                        st.netUsed[num] = true
   594→                                        st.conns[num] = append(st.conns[num],
   595→                                                schematicPinRef{ref: kf.ref, pin: padName})
   596→                                }
   597→                        }
   598→                }
   599→                kf.pads = append(kf.pads, pad)
   600→        }
   601→
   602→        st.footprints = append(st.footprints, kf)
   603→}

   605→// netName resolves the name of a net number, with a deterministic fallback.
   606→func (st *kicadParseState) netName(num int) string {
   607→        if name := st.netNames[num]; name != "" {
   608→                return name
   609→        }
   610→        return fmt.Sprintf("NET_%d", num)
   611→}

   613→// readSegment parses one (segment (start) (end) (width) (layer) (net)) node.
   614→func (st *kicadParseState) readSegment(seg *sNode) {
   615→        start := seg.child("start")
   616→        end := seg.child("end")
   617→        if start == nil || end == nil {
   618→                return
   619→        }
   620→        track := domainlayout.Track{
   621→                Width: 0.25,
   622→                Layer: 0,
   623→                Points: []domainlayout.TrackPoint{
   624→                        {X: sexprFloat(start.arg(0)), Y: sexprFloat(start.arg(1))},
   625→                        {X: sexprFloat(end.arg(0)), Y: sexprFloat(end.arg(1))},
   626→                },
   627→        }
   628→        if w := seg.child("width"); w != nil {
   629→                track.Width = sexprFloat(w.arg(0))
   630→        }
   631→        if layer := seg.child("layer"); layer != nil {
   632→                track.Layer = kicadLayerIndex(layer.arg(0), st.layerCount)
   633→        }
   634→        if net := seg.child("net"); net != nil {
   635→                num := sexprInt(net.arg(0))
   636→                if num > 0 {
   637→                        track.Net = st.netName(num)
   638→                        st.netUsed[num] = true
   639→                }
   640→        }
   641→        st.tracks = append(st.tracks, track)
   642→}

   644→// readVia parses one (via (at) (size) (drill) (layers) (net)) node.
   645→func (st *kicadParseState) readVia(via *sNode) {
   646→        at := via.child("at")
   647→        if at == nil {
   648→                return
   649→        }
   650→        diameter := sexprFloat(via.arg(0)) // forme via micro/blind : diamètre direct
   651→        if size := via.child("size"); size != nil {
   652→                diameter = sexprFloat(size.arg(0))
   653→        }
   654→        drill := 0.0
   655→        if d := via.child("drill"); d != nil {
   656→                drill = sexprFloat(d.arg(0))
   657→        }
   658→        if drill <= 0 {
   659→                drill = diameter / 2
   660→        }
   661→        if diameter <= drill {
   662→                diameter = drill + 0.2
   663→        }
   664→
   665→        fromLayer, toLayer := 0, st.layerCount-1
   666→        if layers := via.child("layers"); layers != nil && len(layers.args()) >= 2 {
   667→                fromLayer = kicadLayerIndex(layers.arg(0), st.layerCount)
   668→                toLayer = kicadLayerIndex(layers.arg(1), st.layerCount)
   669→                if fromLayer == toLayer {
   670→                        toLayer = st.layerCount - 1
   671→                }
   672→        }
   673→
   674→        v := domainlayout.Via{
   675→                X:         sexprFloat(at.arg(0)),
   676→                Y:         sexprFloat(at.arg(1)),
   677→                FromLayer: fromLayer,
   678→                ToLayer:   toLayer,
   679→                Diameter:  diameter,
   680→                Drill:     drill,
   681→        }
   682→        if net := via.child("net"); net != nil {
   683→                num := sexprInt(net.arg(0))
   684→                if num > 0 {
   685→                        v.Net = st.netName(num)
   686→                        st.netUsed[num] = true
   687→                }
   688→        }
   689→        st.vias = append(st.vias, v)
   690→}

   692→// boardOutline derives the board size from the Edge.Cuts rectangles,
   693→// defaulting to 100 x 80 mm.
   694→func (st *kicadParseState) boardOutline(pcb *sNode) (float64, float64) {
   695→        for _, rect := range pcb.childrenWithTag("gr_rect") {
   696→                layer := rect.child("layer")
   697→                if layer != nil && layer.arg(0) != "Edge.Cuts" {
   698→                        continue
   699→                }
   700→                start, end := rect.child("start"), rect.child("end")
   701→                if start == nil || end == nil {
   702→                        continue
   703→                }
   704→                box := geometryAABB{
   705→                        MinX: math.Min(sexprFloat(start.arg(0)), sexprFloat(end.arg(0))),
   706→                        MinY: math.Min(sexprFloat(start.arg(1)), sexprFloat(end.arg(1))),
   707→                        MaxX: math.Max(sexprFloat(start.arg(0)), sexprFloat(end.arg(0))),
   708→                        MaxY: math.Max(sexprFloat(start.arg(1)), sexprFloat(end.arg(1))),
   709→                }
   710→                if st.outline == nil {
   711→                        st.outline = &geometryAABB{}
   712→                        *st.outline = box
   713→                } else {
   714→                        st.outline.union(box)
   715→                }
   716→        }
   717→        if st.outline == nil {
   718→                return 100, 80
   719→        }
   720→        w := st.outline.MaxX - st.outline.MinX
   721→        h := st.outline.MaxY - st.outline.MinY
   722→        if w <= 0 || h <= 0 {
   723→                return 100, 80
   724→        }
   725→        return w, h
   726→}

   728→// kicadBodySize derives the footprint body size from the pad bounding box,
   729→// falling back to 2 x 2 mm.
   730→func kicadBodySize(pads []domainlayout.Pad) (float64, float64) {
   731→        minX, minY, maxX, maxY := 0.0, 0.0, 0.0, 0.0
   732→        for i, p := range pads {
   733→                x0, x1 := p.X-p.Width/2, p.X+p.Width/2
   734→                y0, y1 := p.Y-p.Height/2, p.Y+p.Height/2
   735→                if i == 0 {
   736→                        minX, maxX, minY, maxY = x0, x1, y0, y1
   737→                        continue
   738→                }
   739→                minX = math.Min(minX, x0)
   740→                minY = math.Min(minY, y0)
   741→                maxX = math.Max(maxX, x1)
   742→                maxY = math.Max(maxY, y1)
   743→        }
   744→        if len(pads) == 0 || maxX-minX <= 0 || maxY-minY <= 0 {
   745→                return 2, 2
   746→        }
   747→        return maxX - minX, maxY - minY
   748→}
