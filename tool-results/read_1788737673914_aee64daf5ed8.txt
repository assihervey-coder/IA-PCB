    27→// Règles locales utilisées quand le backend est injoignable (mode démo).
    28→const RULE_MIN_TRACK_WIDTH = 0.2;
    29→const RULE_MIN_VIA_DIAMETER = 0.5;
    30→const RULE_MIN_DRILL = 0.25;
    31→const RULE_EDGE_CLEARANCE = 0.5;
    32→
    33→function computeLocalDRC(layout: LayoutData): DRCResult {
    34→  const violations: DRCViolation[] = [];
    35→  const { width_mm: w, height_mm: h } = layout.board;
    36→  for (const track of layout.tracks) {
    37→    if (track.width < RULE_MIN_TRACK_WIDTH) {
    38→      const p = track.points[0] ?? { x: 0, y: 0 };
    39→      violations.push({
    40→        code: "DRC_TRACK_WIDTH",
    41→        severity: "error",
    42→        message: `Largeur de piste ${track.width.toFixed(2)} mm inférieure au minimum (${RULE_MIN_TRACK_WIDTH.toFixed(2)} mm).`,
    43→        x: p.x,
    44→        y: p.y,
    45→        layer: track.layer,
    46→        net: track.net,
    47→      });
    48→    }
    49→    for (const p of track.points) {
    50→      if (
    51→        p.x < RULE_EDGE_CLEARANCE ||
    52→        p.y < RULE_EDGE_CLEARANCE ||
    53→        p.x > w - RULE_EDGE_CLEARANCE ||
    54→        p.y > h - RULE_EDGE_CLEARANCE
    55→      ) {
    56→        violations.push({
    57→          code: "DRC_EDGE_CLEARANCE",
    58→          severity: "error",
    59→          message: "Piste trop proche du bord de la carte.",
    60→          x: p.x,
    61→          y: p.y,
    62→          layer: track.layer,
    63→          net: track.net,
    64→        });
    65→        break;
    66→      }
    67→    }
    68→  }
    69→  for (const via of layout.vias) {
    70→    if (via.diameter < RULE_MIN_VIA_DIAMETER) {
    71→      violations.push({
    72→        code: "DRC_VIA_DIAMETER",
    73→        severity: "error",
    74→        message: `Diamètre de via ${via.diameter.toFixed(2)} mm inférieur au minimum.`,
    75→        x: via.x,
    76→        y: via.y,
    77→        layer: via.from_layer,
    78→        net: via.net,
    79→      });
    80→    }
    81→    if (via.drill < RULE_MIN_DRILL) {
    82→      violations.push({
    83→        code: "DRC_DRILL",
    84→        severity: "error",
    85→        message: `Forage de via ${via.drill.toFixed(2)} mm inférieur au minimum.`,
    86→        x: via.x,
    87→        y: via.y,
    88→        layer: via.from_layer,
    89→        net: via.net,
    90→      });
    91→    }
    92→  }
    93→  return { passed: violations.length === 0, checked_rules: 4, violations, duration_ms: 8 };
    94→}

    96→/** Placement local déterministe (mode démo) : grille ordonnée, composants fixes conservés. */
    97→function computeLocalPlacement(layout: LayoutData): LayoutData {
    98→  const margin = 6;
    99→  const movable = layout.components.filter((c) => !c.fixed);
   100→  const cols = Math.max(2, Math.ceil(Math.sqrt(movable.length || 1)));
   101→  const rows = Math.max(1, Math.ceil(movable.length / cols));
   102→  const cellW = (layout.board.width_mm - margin * 2) / cols;
   103→  const cellH = (layout.board.height_mm - margin * 2) / rows;
   104→  const placed = new Map<string, { x: number; y: number }>();
   105→  movable.forEach((c, i) => {
   106→    placed.set(c.ref, {
   107→      x: margin + cellW * ((i % cols) + 0.5),
   108→      y: margin + cellH * (Math.floor(i / cols) + 0.5),
   109→    });
   110→  });
   111→  return {
   112→    ...layout,
   113→    components: layout.components.map((c) => {
   114→      const pos = placed.get(c.ref);
   115→      return pos ? { ...c, x: pos.x, y: pos.y } : c;
   116→    }),
   117→  };
   118→}

   120→function roundedRectPath(
   121→  ctx: CanvasRenderingContext2D,
   122→  x: number,
   123→  y: number,
   124→  w: number,
   125→  h: number,
   126→  r: number,
   127→): void {
   128→  const radius = Math.min(r, w / 2, h / 2);
   129→  ctx.beginPath();
   130→  ctx.moveTo(x + radius, y);
   131→  ctx.lineTo(x + w - radius, y);
   132→  ctx.arcTo(x + w, y, x + w, y + radius, radius);
   133→  ctx.lineTo(x + w, y + h - radius);
   134→  ctx.arcTo(x + w, y + h, x + w - radius, y + h, radius);
   135→  ctx.lineTo(x + radius, y + h);
   136→  ctx.arcTo(x, y + h, x, y + h - radius, radius);
   137→  ctx.lineTo(x, y + radius);
   138→  ctx.arcTo(x, y, x + radius, y, radius);
   139→  ctx.closePath();
   140→}

   176→    const ctx = canvas.getContext("2d");
   177→    if (!ctx) return;

   179→    const dpr = window.devicePixelRatio || 1;
   180→    const cssW = canvas.clientWidth || canvasSize.w || 800;
   181→    const cssH = canvas.clientHeight || canvasSize.h || 520;
   182→    if (canvas.width !== Math.round(cssW * dpr) || canvas.height !== Math.round(cssH * dpr)) {
   183→      canvas.width = Math.round(cssW * dpr);
   184→      canvas.height = Math.round(cssH * dpr);
   185→    }
   186→
   187→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   188→    ctx.fillStyle = "#0B1220";
   189→    ctx.fillRect(0, 0, cssW, cssH);
   190→
   191→    const view = viewRef.current;
   192→
   193→    if (!data) return;
   194→    const W = data.board.width_mm;
   195→    const H = data.board.height_mm;
   196→
   197→    // Passer en coordonnées monde (mm).
   198→    ctx.setTransform(dpr * view.scale, 0, 0, dpr * view.scale, dpr * view.offsetX, dpr * view.offsetY);
   199→
   200→    // Plaque.
   201→    ctx.fillStyle = "#0F1B33";
   202→    ctx.fillRect(0, 0, W, H);
   203→
   204→    // Grille 1 mm (+ repères 10 mm) si la densité le permet.
   205→    if (view.scale >= 4.5) {
   206→      ctx.lineWidth = 1 / view.scale;
   207→      ctx.strokeStyle = "rgba(140, 160, 192, 0.10)";
   208→      ctx.beginPath();
   209→      for (let x = 0; x <= W; x += 1) {
   210→        ctx.moveTo(x, 0);
   211→        ctx.lineTo(x, H);
   212→      }
   213→      for (let y = 0; y <= H; y += 1) {
   214→        ctx.moveTo(0, y);
   215→        ctx.lineTo(W, y);
   216→      }
   217→      ctx.stroke();
   218→      ctx.strokeStyle = "rgba(140, 160, 192, 0.20)";
   219→      ctx.beginPath();
   220→      for (let x = 0; x <= W; x += 10) {
   221→        ctx.moveTo(x, 0);
   222→        ctx.lineTo(x, H);
   223→      }
   224→      for (let y = 0; y <= H; y += 10) {
   225→        ctx.moveTo(0, y);
   226→        ctx.lineTo(W, y);
   227→      }
   228→      ctx.stroke();
   229→    }
   230→
   231→    // Contour de la carte.
   232→    ctx.strokeStyle = "#3D5A96";
   233→    ctx.lineWidth = 2 / view.scale;
   234→    ctx.strokeRect(0, 0, W, H);
   235→
   236→    // Pistes.
   237→    ctx.lineJoin = "round";
   238→    ctx.lineCap = "round";
   239→    for (const track of data.tracks) {
   240→      const visible = !hiddenLayers.includes(track.layer);
   241→      let alpha = visible ? 1 : 0.08;
   242→      if (highlightNet && track.net !== highlightNet) alpha = visible ? 0.18 : 0.04;
   243→      ctx.globalAlpha = alpha;
   244→      ctx.strokeStyle = colorForNet(track.net);
   245→      ctx.lineWidth = Math.max(track.width, 1.2 / view.scale);
   246→      ctx.beginPath();
   247→      track.points.forEach((p, i) => (i === 0 ? ctx.moveTo(p.x, p.y) : ctx.lineTo(p.x, p.y)));
   248→      ctx.stroke();
   249→    }
   250→    ctx.globalAlpha = 1;
   251→
   252→    // Vias.
   253→    for (const via of data.vias) {
   254→      const visible = !hiddenLayers.includes(via.from_layer) || !hiddenLayers.includes(via.to_layer);
   255→      ctx.globalAlpha = visible ? 1 : 0.1;
   256→      if (highlightNet && via.net !== highlightNet) ctx.globalAlpha = visible ? 0.2 : 0.06;
   257→      ctx.beginPath();
   258→      ctx.arc(via.x, via.y, via.diameter / 2, 0, Math.PI * 2);
   259→      ctx.fillStyle = "#FFD166";
   260→      ctx.fill();
   261→      ctx.beginPath();
   262→      ctx.arc(via.x, via.y, via.drill / 2, 0, Math.PI * 2);
   263→      ctx.fillStyle = "#0B1220";
   264→      ctx.fill();
   265→    }
   266→    ctx.globalAlpha = 1;
   267→
   268→    // Composants (corps + pastilles).
   269→    for (const comp of data.components) {
   270→      const size = estimateFootprintSize(comp.footprint);
   271→      const isSel = selected?.ref === comp.ref;
   272→      ctx.save();
   273→      ctx.translate(comp.x, comp.y);
   274→      ctx.rotate((comp.rotation * Math.PI) / 180);
   275→      ctx.globalAlpha = highlightNet ? 0.85 : 1;
   276→      roundedRectPath(ctx, -size.w / 2, -size.h / 2, size.w, size.h, 0.5);
   277→      ctx.fillStyle = "#1B2740";
   278→      ctx.fill();
   279→      ctx.strokeStyle = isSel ? "#22D3EE" : "#43598A";
   280→      ctx.lineWidth = (isSel ? 2.2 : 1.2) / view.scale;
   281→      if (isSel) ctx.setLineDash([1.2, 0.8]);
   282→      ctx.stroke();
   283→      ctx.setLineDash([]);
   284→      for (const pad of footprintPadOffsets(size)) {
   285→        ctx.fillStyle = "#F59E0B";
   286→        ctx.fillRect(pad.x - 0.35, pad.y - 0.4, 0.7, 0.8);
   287→      }
   288→      ctx.restore();
   289→      ctx.globalAlpha = 1;
   290→    }
   291→
   292→    // Marqueurs DRC.
   293→    if (drc && !drc.passed) {
   294→      for (const v of drc.violations) {
   295→        ctx.strokeStyle = "#F87171";
   296→        ctx.lineWidth = 0.35;
   297→        ctx.beginPath();
   298→        ctx.arc(v.x, v.y, 1.6, 0, Math.PI * 2);
   299→        ctx.stroke();
   300→        ctx.beginPath();
   301→        ctx.moveTo(v.x - 0.7, v.y - 0.7);
   302→        ctx.lineTo(v.x + 0.7, v.y + 0.7);
   303→        ctx.moveTo(v.x + 0.7, v.y - 0.7);
   304→        ctx.lineTo(v.x - 0.7, v.y + 0.7);
   305→        ctx.stroke();
   306→      }
   307→    }
   308→
   309→    // Repères de référence (espace écran, texte net).
   310→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   311→    ctx.font = "600 11px system-ui, sans-serif";
   312→    ctx.textAlign = "center";
   313→    for (const comp of data.components) {
   314→      const sx = comp.x * view.scale + view.offsetX;
   315→      const sy = comp.y * view.scale + view.offsetY - (estimateFootprintSize(comp.footprint).h / 2) * view.scale - 6;
   316→      ctx.fillStyle = selected?.ref === comp.ref ? "#22D3EE" : "#8CA0C0";
   317→      ctx.fillText(comp.ref, sx, sy);
   318→    }
   319→  }, [hiddenLayers, highlightNet, selected, drc, canvasSize.w, canvasSize.h]);
   320→
   321→  const drawRef = useRef<() => void>(() => {});
   322→  drawRef.current = draw;
   323→
   324→  useEffect(() => {
   325→    draw();
   326→  }, [draw, layout]);
   327→
   328→  // Redimensionnement du canvas.
   329→  useEffect(() => {
   330→    const canvas = canvasRef.current;
   331→    if (!canvas) return;
   332→    const observer = new ResizeObserver(() => {
   333→      setCanvasSize({ w: canvas.clientWidth, h: canvas.clientHeight });
   334→      drawRef.current();
   335→    });
   336→    observer.observe(canvas);
   337→    return () => observer.disconnect();
   338→  }, []);
   339→
   340→  const fitView = useCallback(() => {
   341→    const data = useProjectStore.getState().layout;
   342→    const canvas = canvasRef.current;
   343→    if (!data || !canvas) return;
   344→    const cssW = canvas.clientWidth || 800;
   345→    const cssH = canvas.clientHeight || 520;
   346→    const scale = Math.min(
   347→      (cssW - FIT_MARGIN * 2) / data.board.width_mm,
   348→      (cssH - FIT_MARGIN * 2) / data.board.height_mm,
   349→    );
   350→    viewRef.current = {
   351→      scale,
   352→      offsetX: (cssW - data.board.width_mm * scale) / 2,
   353→      offsetY: (cssH - data.board.height_mm * scale) / 2,
   354→    };
   355→    fitScaleRef.current = scale;
   356→    setZoomLabel(100);
   357→    drawRef.current();
   358→  }, []);
   359→
   360→  // Ajustement initial dès que le layout est disponible.

   362→  useEffect(() => {
   363→    if (!layout || !canvasRef.current) return;
   364→    const key = `${currentProject?.id ?? "demo"}-${layout.board.width_mm}x${layout.board.height_mm}`;
   365→    if (fittedFor.current !== key) {
   366→      fittedFor.current = key;
   367→      fitView();
   368→    }
   369→  }, [layout, currentProject, fitView]);
   370→
   371→  // Zoom molette centré sur le curseur.
   372→  useEffect(() => {
   373→    const canvas = canvasRef.current;
   374→    if (!canvas) return;
   375→    const onWheel = (e: WheelEvent) => {
   376→      e.preventDefault();
   377→      const rect = canvas.getBoundingClientRect();
   378→      const mx = e.clientX - rect.left;
   379→      const my = e.clientY - rect.top;
   380→      const view = viewRef.current;
   381→      const before = screenToWorld(mx, my, view);
   382→      const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
   383→      view.scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, view.scale * factor));
   384→      view.offsetX = mx - before.x * view.scale;
   385→      view.offsetY = my - before.y * view.scale;
   386→      if (fitScaleRef.current > 0) {
   387→        setZoomLabel(Math.round((view.scale / fitScaleRef.current) * 100));
   388→      }
   389→      drawRef.current();
   390→    };
   391→    canvas.addEventListener("wheel", onWheel, { passive: false });
   392→    return () => canvas.removeEventListener("wheel", onWheel);
   393→  }, []);
   394→
   395→  const zoomBy = (factor: number) => {
   396→    const view = viewRef.current;
   397→    const canvas = canvasRef.current;
   398→    if (!canvas) return;
   399→    const cx = canvas.clientWidth / 2;
   400→    const cy = canvas.clientHeight / 2;
   401→    const before = screenToWorld(cx, cy, view);
   402→    view.scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, view.scale * factor));
   403→    view.offsetX = cx - before.x * view.scale;
   404→    view.offsetY = cy - before.y * view.scale;
   405→    setZoomLabel(Math.round((view.scale / fitScaleRef.current) * 100));
   406→    drawRef.current();
   407→  };
   408→
   409→  // ------- Interactions souris -------
   410→  const handlePointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   411→    if (e.button !== 0) return;
   412→    canvasRef.current?.setPointerCapture(e.pointerId);
   413→    dragRef.current = { mx: e.clientX, my: e.clientY, ox: viewRef.current.offsetX, oy: viewRef.current.offsetY, moved: false };
   414→  };
   415→
   416→  const handlePointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
   417→    const drag = dragRef.current;
   418→    if (!drag) return;
   419→    const dx = e.clientX - drag.mx;
   420→    const dy = e.clientY - drag.my;
   421→    if (Math.abs(dx) + Math.abs(dy) > 4) drag.moved = true;
   422→    if (drag.moved) {
   423→      viewRef.current.offsetX = drag.ox + dx;
   424→      viewRef.current.offsetY = drag.oy + dy;
   425→      drawRef.current();
   426→    }
   427→  };
   428→
   429→  const handlePointerUp = (e: React.PointerEvent<HTMLCanvasElement>) => {
   430→    const drag = dragRef.current;
   431→    dragRef.current = null;
   432→    if (!drag || drag.moved) return;
   433→    const canvas = canvasRef.current;
   434→    if (!canvas) return;
   435→    const rect = canvas.getBoundingClientRect();
   436→    const world = screenToWorld(e.clientX - rect.left, e.clientY - rect.top, viewRef.current);
   437→    const data = useProjectStore.getState().layout;
   438→    if (!data) return;
   439→    let hit: LayoutComponent | null = null;
   440→    for (let i = data.components.length - 1; i >= 0; i--) {
   441→      const c = data.components[i];
   442→      const size = estimateFootprintSize(c.footprint);
   443→      const swap = Math.abs(c.rotation % 180) === 90;
   444→      const hw = (swap ? size.h : size.w) / 2 + 0.4;
   445→      const hh = (swap ? size.w : size.h) / 2 + 0.4;
   446→      if (Math.abs(world.x - c.x) <= hw && Math.abs(world.y - c.y) <= hh) {
   447→        hit = c;
   448→        break;
   449→      }
   450→    }
   451→    setSelected(hit);
   452→  };
   453→
   454→  // ------- Actions -------
   455→  const handlePlace = async () => {
   456→    if (!projectId || busyAction) return;
   457→    setBusyAction("place");
   458→    try {
   459→      await api.startPlace(projectId, "heuristic");
   460→      await refreshLayout(projectId);
   461→      pushToast("Placement IA terminé, layout mis à jour.", "success");
   462→    } catch (err) {
   463→      if (isBackendDown(err)) {
   464→        const data = useProjectStore.getState().layout;
   465→        if (data) {
   466→          useProjectStore.setState({ layout: computeLocalPlacement(data) });
   467→          pushToast("Mode démo : placement local simulé (grille ordonnée).", "info");
   468→        }
   469→      } else {
   470→        pushToast("Échec du placement IA.", "error");
   471→      }
   472→    } finally {
   473→      setBusyAction(null);
   474→    }
   475→  };
   476→
   477→  const handleSave = async () => {
   478→    const data = useProjectStore.getState().layout;
   479→    if (!projectId || !data || busyAction) return;
   480→    setBusyAction("save");
   481→    try {
   482→      await api.putLayout(projectId, data);
   483→      pushToast("Layout enregistré sur le backend.", "success");
   484→    } catch (err) {
   485→      if (isBackendDown(err)) {
   486→        pushToast("Mode démo : enregistrement local uniquement.", "info");
   487→      } else {
   488→        pushToast("Échec de l'enregistrement du layout.", "error");
   489→      }
   490→    } finally {
   491→      setBusyAction(null);
   492→    }
   493→  };
   494→
   495→  const handleDrc = async () => {
   496→    if (busyAction) return;
   497→    const data = useProjectStore.getState().layout;
   498→    if (!projectId || !data) return;
   499→    setBusyAction("drc");
   500→    try {
   501→      const result = await api.runDRC(projectId);
   502→      setDRC(result);
   503→      pushToast(
   504→        result.passed
   505→          ? `DRC conforme (${result.checked_rules} règles vérifiées).`
   506→          : `DRC : ${result.violations.length} violation(s).`,
   507→        result.passed ? "success" : "error",
   508→      );
   509→    } catch (err) {
   510→      if (isBackendDown(err)) {
   511→        const local = computeLocalDRC(data);
   512→        setDRC(local);
   513→        pushToast(
   514→          local.passed
   515→            ? "Mode démo : DRC local conforme."
   516→            : `Mode démo : DRC local — ${local.violations.length} violation(s).`,
   517→          local.passed ? "success" : "error",
   518→        );
   519→      } else {
   520→        pushToast("Échec de l'exécution du DRC.", "error");
   521→      }
   522→    } finally {
   523→      setBusyAction(null);
   524→    }
   525→  };
   526→
   527→  const applyProperties = async (patch: ComponentPropertiesPatch) => {
   528→    const data = useProjectStore.getState().layout;
   529→    if (!data || !selected) return;
   530→    const next: LayoutData = {
   531→      ...data,
   532→      components: data.components.map((c) =>
   533→        c.ref === selected.ref
   534→          ? {
   535→              ...c,
   536→              footprint: patch.footprint,
   537→              x: patch.x,
   538→              y: patch.y,
   539→              rotation: patch.rotation,
   540→              fixed: patch.fixed,
   541→            }
   542→          : c,
   543→      ),
   544→    };
   545→    useProjectStore.setState({ layout: next });
   546→    setSelected(null);
   547→    if (projectId && !demoMode) {
   548→      try {
   549→        await api.putLayout(projectId, next);
   550→        pushToast("Composant mis à jour et enregistré.", "success");
   551→      } catch {
   552→        pushToast("Composant mis à jour localement (backend injoignable).", "info");
   553→      }
   554→    }
   555→  };
   556→
   557→  const toggleLayer = (index: number) => {
   558→    setHiddenLayers((prev) => (prev.includes(index) ? prev.filter((l) => l !== index) : [...prev, index]));
   559→  };
   560→
   561→  const query = projectId ? `?project=${encodeURIComponent(projectId)}` : "";
   562→  const layerNames = layout?.board.layer_names ?? [];
   563→
   564→  return (
   565→    <div className="page">
   566→      <div className="page-header">
   567→        <div>
   568→          <h1 className="page-title">Éditeur PCB</h1>
   569→          <p className="page-subtitle">
   570→            {currentProject?.name ?? "Projet sans nom"}
   571→            {projectId ? ` — projet ${projectId}` : " — données de démonstration"}
   572→          </p>
   573→        </div>
   574→      </div>
   575→
   576→      {demoMode ? (
   577→        <div className="demo-banner" data-testid="demo-banner" role="status">
   578→          Mode démo : backend injoignable, carte d&apos;exemple affichée.
   579→        </div>
   580→      ) : null}
   581→
   582→      <div className="editor-layout" style={{ gridTemplateColumns: "270px 1fr" }}>
   583→        <aside className="side-panel">
   584→          <div className="side-box panel">
   585→            <h3 className="panel-title">Couches</h3>
   586→            <div className="layer-list">
   587→              {layerNames.map((name, index) => (
   588→                <label key={name} className="layer-item">
   589→                  <input
   590→                    type="checkbox"
   591→                    checked={!hiddenLayers.includes(index)}
   592→                    onChange={() => toggleLayer(index)}
   593→                  />
   594→                  <span>{name}</span>
   595→                </label>
   596→              ))}
   597→            </div>
   598→          </div>
   599→
   600→          <div className="side-box panel">
   601→            <h3 className="panel-title">Nets ({layout?.nets.length ?? 0})</h3>
   602→            <ul className="net-list">
   603→              {(layout?.nets ?? []).map((net) => (
   604→                <li
   605→                  key={net.name}
   606→                  className={`net-item${highlightNet === net.name ? " active" : ""}`}
   607→                  onClick={() => setHighlightNet((prev) => (prev === net.name ? null : net.name))}
   608→                  title="Cliquer pour isoler ce net"
   609→                >
   610→                  <span className="net-dot" style={{ background: colorForNet(net.name) }} />
   611→                  <span className="net-name">{net.name}</span>
   612→                  <span className="net-count">{net.pad_count} pads</span>
   613→                </li>
   614→              ))}
   615→            </ul>
   616→          </div>
   617→
   618→          <div className="side-box panel">
   619→            <h3 className="panel-title">Légende</h3>
   620→            <div className="legend">
   621→              <div className="legend-item">
   622→                <span className="legend-swatch" style={{ background: "#F59E0B" }} />
   623→                Pastilles / cuivre F.Cu
   624→              </div>
   625→              <div className="legend-item">
   626→                <span className="legend-swatch" style={{ background: "#FFD166" }} />
   627→                Vias (couche B en or)
   628→              </div>
   629→              <div className="legend-item">
   630→                <span className="legend-swatch" style={{ background: "#22D3EE" }} />
   631→                Composant sélectionné
   632→              </div>
   633→              <div className="legend-item">
   634→                <span className="legend-swatch" style={{ background: "#F87171" }} />
   635→                Violation DRC
   636→              </div>
   637→            </div>
   638→          </div>
   639→        </aside>
   640→
   641→        <section className="canvas-wrap panel">
   642→          <div className="toolbar">
   643→            <Button size="sm" data-testid="ai-place-btn" onClick={() => void handlePlace()} disabled={busyAction !== null}>
   644→              {busyAction === "place" ? "Placement…" : "Placement IA"}
   645→            </Button>
   646→            <Link className="btn btn-secondary btn-sm" href={`/pages/pcb-layout/router${query}`}>
   647→              Routage IA
   648→            </Link>
   649→            <Button size="sm" data-testid="drc-run-btn" onClick={() => void handleDrc()} disabled={busyAction !== null}>
   650→              {busyAction === "drc" ? "Analyse…" : "DRC"}
   651→            </Button>
   652→            <Link className="btn btn-secondary btn-sm" href={`/pages/schematic-editor${query}`}>
   653→              ERC
   654→            </Link>
   655→            <span className="toolbar-sep" />
   656→            <Link className="btn btn-secondary btn-sm" href={`/pages/export${query}`}>
   657→              Exporter
   658→            </Link>
   659→            <Link className="btn btn-secondary btn-sm" href={`/pages/pcb-layout/viewer${query}`}>
   660→              Vue 3D
   661→            </Link>
   662→            <span className="toolbar-sep" />
   663→            <Button size="sm" variant="ghost" onClick={() => void handleSave()} disabled={busyAction !== null}>
   664→              {busyAction === "save" ? "Enregistrement…" : "Enregistrer"}
   665→            </Button>
   666→            <span className="toolbar-sep" />
   667→            <Button size="sm" variant="secondary" onClick={() => zoomBy(1.2)} aria-label="Zoom avant">
   668→              Zoom +
   669→            </Button>
   670→            <Button size="sm" variant="secondary" onClick={() => zoomBy(1 / 1.2)} aria-label="Zoom arrière">
   671→              Zoom −
   672→            </Button>
   673→            <Button size="sm" variant="ghost" onClick={fitView}>
   674→              Ajuster
   675→            </Button>
   676→            <span className="zoom-label">{percentFormat(zoomLabel)}</span>
   677→          </div>
   678→
   679→          <canvas
   680→            ref={canvasRef}
   681→            data-testid="pcb-canvas"
   682→            className="pcb-canvas"
   683→            onPointerDown={handlePointerDown}
   684→            onPointerMove={handlePointerMove}
   685→            onPointerUp={handlePointerUp}
   686→            onPointerLeave={handlePointerUp}
   687→          />
   688→
   689→          {!layout ? (
   690→            <div className="hint-empty">
   691→              Aucune donnée de layout. Créez un projet, importez un fichier ou lancez le placement IA.
   692→            </div>
   693→          ) : null}
   694→
   695→          <div className="canvas-status">
   696→            <span>
   697→              Carte : {layout ? `${formatMm(layout.board.width_mm, 0)} × ${formatMm(layout.board.height_mm, 0)}` : "—"}
   698→            </span>
   699→            <span>{layout?.tracks.length ?? 0} pistes — {layout?.vias.length ?? 0} vias</span>
   700→            {highlightNet ? <span>Net isolé : {highlightNet}</span> : null}
   701→            {selected ? <span>Sélection : {selected.ref}</span> : <span className="muted">Cliquez sur un composant</span>}
   702→          </div>
   703→        </section>
   704→      </div>
   705→
   706→      {drc ? (
   707→        <section className="erc-panel panel">
   708→          <div className="erc-header">
   709→            <h3>Résultats DRC</h3>
   710→            <div className="page-actions">
   711→              <span className="erc-summary">
   712→                {drc.checked_rules} règles vérifiées en {drc.duration_ms} ms
   713→              </span>
   714→              <Button size="sm" variant="ghost" onClick={() => setDRC(null)}>
   715→                Masquer
   716→              </Button>
   717→            </div>
   718→          </div>
   719→          <div className={`pass-banner ${drc.passed ? "ok" : "fail"}`}>
   720→            {drc.passed
   721→              ? "DRC conforme : aucune violation détectée."
   722→              : `DRC : ${drc.violations.length} violation(s) — marquées en rouge sur la carte.`}
   723→          </div>
   724→          {drc.violations.length > 0 ? (
   725→            <ul className="violation-list" style={{ marginTop: 10 }}>
   726→              {drc.violations.map((v, i) => (
   727→                <li key={i} className="violation-item">
   728→                  <span className={`sev sev-${v.severity}`}>
   729→                    {v.severity === "error" ? "Erreur" : "Avertissement"}
   730→                  </span>
   731→                  <div>
   732→                    <strong>{v.code}</strong>
   733→                    <span> — {v.message}</span>
   734→                    <div className="violation-loc">
   735→                      Net {v.net || "—"} · Couche {layerNames[v.layer] ?? v.layer} · x {formatMm(v.x)} · y {formatMm(v.y)}
   736→                    </div>
   737→                  </div>
   738→                </li>
   739→              ))}
   740→            </ul>
   741→          ) : null}
   742→        </section>
   743→      ) : null}
   744→
   745→      {selected ? (
   746→        <Modal title={`Propriétés — ${selected.ref}`} onClose={() => setSelected(null)}>
   747→          <ComponentPropertiesForm
   748→            component={selected}
   749→            onSubmit={(patch) => void applyProperties(patch)}
   750→            onCancel={() => setSelected(null)}
   751→          />
   752→        </Modal>
   753→      ) : null}
   754→    </div>
   755→  );
   756→}
