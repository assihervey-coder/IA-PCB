     1→     1→"""Grid-based PCB routing environment with a deterministic A* fallback.
     2→     2→
     3→     3→The environment models the routing of ONE net at a time on a regular grid
     4→     4→extracted from the board description. It follows the gym-like API::
     5→     5→
     6→     6→    env = PCBRouteEnv(board, nets, EnvConfig(...))
     7→     7→    obs = env.reset(net_index)
     8→     8→    obs, reward, terminated, truncated, info = env.step(action)
     9→     9→
    10→    10→Grid and coordinates
    11→    11→--------------------
    12→    12→All input/output coordinates are millimetres with the origin at the
    13→    13→top-left corner of the board; layer 0 is F.Cu. Internally the board is
    14→    14→discretised on a regular grid of resolution ``res`` (the board's
    15→    15→``grid_resolution_mm`` when provided, otherwise ``EnvConfig.grid_mm``):
    16→    16→cell ``(gx, gy)`` is centred at ``(gx * res, gy * res)`` and internal
    17→    17→cells are ``(x, y, layer)`` tuples with numpy arrays indexed ``[l, y, x]``.
    18→    18→
    19→    19→Blocking rules
    20→    20→--------------
    21→    21→* board border: a ring of ``clearance_cells`` cells is blocked;
    22→    22→* pads of OTHER nets: the pad cell plus a square halo of
    23→    23→  ``clearance_cells`` cells (on the pad's own layer) is blocked;
    24→    24→* pads of the CURRENT net: terminal cells (free, they are the goals);
    25→    25→* routes registered through :meth:`set_routes` (or by previous successful
    26→    26→  router calls): blocked for every other net.
    27→    27→
    28→    28→Observation
    29→    29→-----------
    30→    30→``float32`` array of shape ``(layer_count + 3, H, W)``:
    31→    31→
    32→    32→* channels ``0 .. layer_count-1``: obstacle plane per layer;
    33→    33→* channel ``layer_count``: source mask (episode start pad cell);
    34→    34→* channel ``layer_count + 1``: target mask (remaining target pad cells);
    35→    35→* channel ``layer_count + 2``: layer indicator (constant plane equal to
    36→    36→  ``current_layer / max(1, layer_count - 1)``).
    37→    37→
    38→    38→Actions
    39→    39→-------
    40→    40→12 discrete actions from :class:`~src.environment.action_space.ActionSpace`
    41→    41→(4 moves x {stay, via-up, via-down}). An invalid move (blocked cell, off
    42→    42→grid, via onto a blocked layer cell) is a collision: the agent stays in
    43→    43→place and receives a negative reward.
    44→    44→
    45→    45→Success
    46→    46→-------
    47→    47→The episode terminates successfully as soon as the agent reaches any
    48→    48→remaining target pad cell of the net. The deterministic fallback router
    49→    49→:meth:`astar_route` handles full multi-terminal nets by chaining A* legs
    50→    50→between pads (nearest-terminal-first), which is what the gRPC service uses
    51→    51→when no RL model is available.
    52→    52→"""
    53→    53→

    68→    68→
    69→    69→_MOVES_4: tuple[tuple[int, int], ...] = ((0, -1), (1, 0), (0, 1), (-1, 0))
    70→    70→
    71→    71→
    72→    72→@dataclass
    73→    73→class EnvConfig:
    74→    74→    """Environment / router configuration.
    75→    75→
    76→    76→    Attributes:
    77→    77→        grid_mm: default grid resolution (mm) when the board does not
    78→    78→            provide ``grid_resolution_mm``.
    79→    79→        max_steps: episode step limit; ``None`` selects an automatic limit
    80→    80→            of ``max(64, 4 * (W + H) * layer_count)`` steps.
    81→    81→        via_penalty: A* cost (in cell units) and reward penalty of a via.
    82→    82→        step_penalty: reward penalty per step.

    84→    84→        success_bonus: terminal reward for reaching a target pad.
    85→    85→        collision_penalty: reward penalty for invalid moves.

    87→    87→            and around the board border.

    89→    89→            stochastic extensions; routing itself is fully deterministic).

   145→   145→class PCBRouteEnv:
   146→   146→    """Gym-like single-net PCB routing environment + deterministic A* router."""
   147→   147→
   148→   148→    def __init__(
   149→   149→        self,
   150→   150→        board: dict,
   151→   151→        nets: list[dict],
   152→   152→        config: EnvConfig | None = None,
   153→   153→    ) -> None:
   154→   154→        """Build the grid environment from plain-Python descriptions.
   155→   155→
   156→   156→        Args:
   157→   157→            board: dict with ``width_mm``, ``height_mm``, ``layer_count``,
   158→   158→                ``grid_resolution_mm`` (optional) and ``layer_names``.
   159→   159→            nets: list of dicts with ``name``, ``net_class``,
   160→   160→                ``pads`` (list of dicts with ``component_ref``, ``pad_name``,
   161→   161→                ``position`` ``{x, y}``, ``layer``, ``width_mm``,
   162→   162→                ``height_mm``), ``min_track_width_mm`` and ``clearance_mm``.
   163→   163→            config: optional :class:`EnvConfig` override.
   164→   164→        """
   165→   165→        self.cfg = config if config is not None else EnvConfig()
   166→   166→        self.rng = random.Random(self.cfg.seed)
   167→   167→
   168→   168→        self.width_mm = float(board.get("width_mm", 100.0) or 100.0)
   169→   169→        self.height_mm = float(board.get("height_mm", 100.0) or 100.0)
   170→   170→        self.layer_count = max(1, int(board.get("layer_count", 2) or 2))
   171→   171→        names = board.get("layer_names")
   172→   172→        self.layer_names: list[str] = list(names) if names else [
   173→   173→            f"L{i}" for i in range(self.layer_count)
   174→   174→        ]
   175→   175→        if len(self.layer_names) < self.layer_count:
   176→   176→            self.layer_names += [f"L{i}" for i in range(len(self.layer_names), self.layer_count)]
   177→   177→
   178→   178→        res = float(board.get("grid_resolution_mm") or 0.0)
   179→   179→        if res <= 0.0:
   180→   180→            res = float(self.cfg.grid_mm)
   181→   181→        self.res: float = max(res, 1e-6)
   182→   182→
   183→   183→        self.grid_w: int = max(3, int(round(self.width_mm / self.res)) + 1)
   184→   184→        self.grid_h: int = max(3, int(round(self.height_mm / self.res)) + 1)
   185→   185→
   186→   186→        self.action_space = ActionSpace()
   187→   187→        self.reward_shaper = RewardShaper(
   188→   188→            RewardConfig(
   189→   189→                via_penalty=self.cfg.via_penalty,
   190→   190→                step_penalty=self.cfg.step_penalty,
   191→   191→                progress_coef=self.cfg.progress_coef,
   192→   192→                success_bonus=self.cfg.success_bonus,
   193→   193→                collision_penalty=self.cfg.collision_penalty,
   194→   194→            )
   195→   195→        )
   196→   196→
   197→   197→        self._nets: list[_Net] = [self._parse_net(n) for n in (nets or [])]
   198→   198→        self._routes_cells: dict[str, set[Cell]] = {}  # per-net routed cells
   199→   199→        self._net_routes: dict[str, dict] = {}         # per-net route dict output
   200→   200→        self._episode = _Episode()
   201→   201→        self.max_steps: int = self._resolve_max_steps()
   202→   202→
   203→   203→        self._static: np.ndarray | None = None  # lazily built blocking grid
   204→   204→        self._build_static_blocking()

   208→   208→    def _parse_net(self, spec: dict) -> _Net:
   209→   209→        """Convert a net description dict into the internal representation."""
   210→   210→        pads: list[_Pad] = []
   211→   211→        for pad in spec.get("pads") or []:
   212→   212→            pos = pad.get("position") or {}
   213→   213→            layer = int(pad.get("layer", 0) or 0)
   214→   214→            layer = min(max(layer, 0), self.layer_count - 1)
   215→   215→            pads.append(
   216→   216→                _Pad(
   217→   217→                    component_ref=str(pad.get("component_ref", "") or ""),
   218→   218→                    pad_name=str(pad.get("pad_name", "") or ""),
   219→   219→                    x_mm=float(pos.get("x", 0.0) or 0.0),
   220→   220→                    y_mm=float(pos.get("y", 0.0) or 0.0),
   221→   221→                    layer=layer,
   222→   222→                    width_mm=float(pad.get("width_mm", 0.3) or 0.3),
   223→   223→                    height_mm=float(pad.get("height_mm", 0.3) or 0.3),
   224→   224→                )
   225→   225→            )
   226→   226→        return _Net(
   227→   227→            name=str(spec.get("name", f"net{len(pads)}") or f"net{len(pads)}"),
   228→   228→            net_class=str(spec.get("net_class", "default") or "default"),
   229→   229→            pads=pads,
   230→   230→            min_track_width_mm=float(spec.get("min_track_width_mm", 0.25) or 0.25),
   231→   231→            clearance_mm=float(spec.get("clearance_mm", 0.2) or 0.2),
   232→   232→        )
   233→   233→
   234→   234→    def _resolve_max_steps(self) -> int:
   235→   235→        """Compute the episode step limit."""
   236→   236→        if self.cfg.max_steps is not None and self.cfg.max_steps > 0:
   237→   237→            return int(self.cfg.max_steps)
   238→   238→        return max(64, 4 * (self.grid_w + self.grid_h) * self.layer_count)
   239→   239→
   240→   240→    def _build_static_blocking(self) -> None:
   241→   241→        """Build the static blocking grid: border margin + all pad halos."""
   242→   242→        layers, height, width = self.layer_count, self.grid_h, self.grid_w
   243→   243→        margin = max(0, int(self.cfg.clearance_cells))
   244→   244→        for attempt_margin in (margin, 0):
   245→   245→            grid = np.zeros((layers, height, width), dtype=bool)
   246→   246→            if attempt_margin > 0:
   247→   247→                grid[:, :attempt_margin, :] = True
   248→   248→                grid[:, height - attempt_margin:, :] = True
   249→   249→                grid[:, :, :attempt_margin] = True
   250→   250→                grid[:, :, width - attempt_margin:] = True
   251→   251→            for net in self._nets:
   252→   252→                for pad in net.pads:
   253→   253→                    gx, gy = self.to_cell(pad.x_mm, pad.y_mm)
   254→   254→                    for dy in range(-attempt_margin, attempt_margin + 1):
   255→   255→                        for dx in range(-attempt_margin, attempt_margin + 1):
   256→   256→                            x, y = gx + dx, gy + dy
   257→   257→                            if 0 <= x < width and 0 <= y < height:
   258→   258→                                grid[pad.layer, y, x] = True
   259→   259→            if bool((~grid).any()) or attempt_margin == 0:
   260→   260→                self._static = grid
   261→   261→                return
   262→   262→        self._static = grid
   263→   263→

   266→   266→    def to_cell(self, x_mm: float, y_mm: float) -> tuple[int, int]:
   267→   267→        """Convert millimetre coordinates to integer grid cell indices."""
   268→   268→        gx = min(max(int(round(float(x_mm) / self.res)), 0), self.grid_w - 1)
   269→   269→        gy = min(max(int(round(float(y_mm) / self.res)), 0), self.grid_h - 1)
   270→   270→        return gx, gy
   271→   271→
   272→   272→    def to_mm(self, gx: int, gy: int) -> tuple[float, float]:
   273→   273→        """Convert grid cell indices to millimetre coordinates."""
   274→   274→        return float(gx) * self.res, float(gy) * self.res

   276→   276→    def _pad_cell(self, pad: _Pad) -> Cell:
   277→   277→        """Grid cell of a pad (on the pad's own layer)."""
   278→   278→        gx, gy = self.to_cell(pad.x_mm, pad.y_mm)
   279→   279→        return (gx, gy, pad.layer)

   281→   281→    @staticmethod
   282→   282→    def _pad_dist(a: Cell, b: Cell) -> int:
   283→   283→        """Manhattan distance between cells, counting a layer change as 1."""
   284→   284→        return abs(a[0] - b[0]) + abs(a[1] - b[1]) + (1 if a[2] != b[2] else 0)

   314→   314→
   315→   315→    def _free_mask(self, net_index: int) -> np.ndarray:
   316→   316→        """Boolean free-cell mask ``(L, H, W)`` for the given net.
   317→   317→
   318→   318→        Static blocking (border + all pad halos) minus the current net's own
   319→   319→        pads and their halo, minus routes of the current net, plus routes of
   320→   320→        the other nets as obstacles.
   321→   321→        """
   322→   322→        layers, height, width = self.layer_count, self.grid_h, self.grid_w
   323→   323→        free = ~self._static  # copy through negation
   324→   324→        net = self._nets[net_index]
   325→   325→        margin = max(0, int(self.cfg.clearance_cells))
   326→   326→        # Unblock own pads and their clearance halo (own net may route there).
   327→   327→        for pad in net.pads:
   328→   328→            gx, gy = self.to_cell(pad.x_mm, pad.y_mm)
   329→   329→            for dy in range(-margin, margin + 1):
   330→   330→                for dx in range(-margin, margin + 1):
   331→   331→                    x, y = gx + dx, gy + dy
   332→   332→                    if 0 <= x < width and 0 <= y < height:
   333→   333→                        free[pad.layer, y, x] = True
   334→   334→        # Block routes belonging to other nets.
   335→   335→        for name, cells in self._routes_cells.items():
   336→   336→            if name == net.name:
   337→   337→                continue
   338→   338→            for (x, y, layer) in cells:
   339→   339→                if 0 <= x < width and 0 <= y < height and 0 <= layer < layers:
   340→   340→                    free[layer, y, x] = False
   341→   341→        return free
   342→   342→
   343→   343→    def _observation(self) -> np.ndarray:
   344→   344→        """Build the multi-channel float32 observation for the current state."""
   345→   345→        channels = self.layer_count + 3
   346→   346→        obs = np.zeros((channels, self.grid_h, self.grid_w), dtype=np.float32)
   347→   347→        ep = self._episode
   348→   348→        if 0 <= ep.net_index < len(self._nets):
   349→   349→            free = self._free_mask(ep.net_index)
   350→   350→            obs[: self.layer_count] = ~free
   351→   351→        if ep.start is not None:
   352→   352→            obs[self.layer_count, ep.start[1], ep.start[0]] = 1.0
   353→   353→        for (tx, ty, _tl) in ep.targets:
   354→   354→            obs[self.layer_count + 1, ty, tx] = 1.0
   355→   355→        obs[self.layer_count + 2, :, :] = ep.layer / max(1, self.layer_count - 1)
   356→   356→        return obs
   357→   357→
   358→   358→    def _nearest_target_distance(self) -> int:
   359→   359→        """Manhattan distance (cells) to the nearest remaining target."""
   360→   360→        ep = self._episode
   361→   361→        if not ep.targets:
   362→   362→            return 0
   363→   363→        here = (ep.x, ep.y, ep.layer)
   364→   364→        return min(self._pad_dist(here, t) for t in ep.targets)
   365→   365→
   366→   366→    # ----------------------------------------------------------------- episode
   367→   367→
   368→   368→    def _pick_start(self, cells: Sequence[Cell]) -> Cell:
   369→   369→        """Pick the pad cell nearest to the centroid of all pad cells."""
   370→   370→        cx = sum(c[0] for c in cells) / float(len(cells))
   371→   371→        cy = sum(c[1] for c in cells) / float(len(cells))
   372→   372→        return min(cells, key=lambda c: (abs(c[0] - cx) + abs(c[1] - cy), c))
   373→   373→
   374→   374→    def reset(self, net_index: int) -> np.ndarray:
   375→   375→        """Start a new episode routing ``nets[net_index]``.
   376→   376→
   377→   377→        Args:
   378→   378→            net_index: index of the net to route.
   379→   379→
   380→   380→        Returns:
   381→   381→            The initial observation (``float32``, shape ``observation_shape``).
   382→   382→
   383→   383→        Raises:
   384→   384→            IndexError: if ``net_index`` is out of range.
   385→   385→        """
   386→   386→        if not 0 <= net_index < len(self._nets):
   387→   387→            raise IndexError(f"net_index {net_index} out of range [0, {len(self._nets)})")
   388→   388→        net = self._nets[net_index]
   389→   389→        cells: list[Cell] = []
   390→   390→        seen: set[Cell] = set()
   391→   391→        for pad in net.pads:
   392→   392→            cell = self._pad_cell(pad)
   393→   393→            if cell not in seen:
   394→   394→                seen.add(cell)
   395→   395→                cells.append(cell)
   396→   396→
   397→   397→        start = self._pick_start(cells) if cells else None
   398→   398→        ep = self._episode
   399→   399→        ep.net_index = net_index
   400→   400→        ep.steps = 0
   401→   401→        ep.vias = 0
   402→   402→        ep.start = start
   403→   403→        ep.targets = [c for c in cells if c != start]
   404→   404→        ep.path = [start] if start else []
   405→   405→        ep.visited = {start} if start else set()
   406→   406→        ep.done = False
   407→   407→        ep.success = False
   408→   408→        if start is not None:
   409→   409→            ep.x, ep.y, ep.layer = start
   410→   410→        return self._observation()
   411→   411→
   412→   412→    def step(self, action: int) -> tuple[np.ndarray, float, bool, bool, dict]:
   413→   413→        """Apply one action.
   414→   414→
   415→   415→        Args:
   416→   416→            action: index in ``[0, 12)`` from :class:`ActionSpace`.
   417→   417→
   418→   418→        Returns:
   419→   419→            ``(obs, reward, terminated, truncated, info)``. ``terminated`` is
   420→   420→            True on success (a target pad cell was reached) and ``truncated``
   421→   421→            on the step budget exhaustion. Invalid moves are collisions: the
   422→   422→            agent stays in place with a negative reward.
   423→   423→
   424→   424→        Raises:
   425→   425→            IndexError: if ``action`` is out of range.
   426→   426→        """
   427→   427→        if not self.action_space.contains(action):
   428→   428→            raise IndexError(f"Action {action} out of range [0, {self.action_space.n})")
   429→   429→        ep = self._episode
   430→   430→        info = self._info()
   431→   431→        if ep.done:  # idempotent post-terminal step
   432→   432→            return self._observation(), 0.0, ep.success, False, info
   433→   433→
   434→   434→        dx, dy, dlayer = self.action_space.decode(int(action))
   435→   435→        shaper = self.reward_shaper
   436→   436→        reward = shaper.step_cost()
   437→   437→        ep.steps += 1
   438→   438→
   439→   439→        prev_dist = self._nearest_target_distance()
   440→   440→
   441→   441→        if dlayer != 0:
   442→   442→            new_layer = ep.layer + dlayer
   443→   443→            valid_layer = 0 <= new_layer < self.layer_count
   444→   444→            cell_free = (
   445→   445→                valid_layer
   446→   446→                and bool(self._free_mask(ep.net_index)[new_layer, ep.y, ep.x])
   447→   447→            )
   448→   448→            if not valid_layer or not cell_free:
   449→   449→                reward += shaper.collision()
   450→   450→            else:
   451→   451→                ep.layer = new_layer
   452→   452→                ep.vias += 1
   453→   453→                reward += shaper.via_cost()
   454→   454→                cell = (ep.x, ep.y, ep.layer)
   455→   455→                ep.visited.add(cell)
   456→   456→                ep.path.append(cell)
   457→   457→        else:
   458→   458→            nx, ny = ep.x + dx, ep.y + dy
   459→   459→            in_bounds = 0 <= nx < self.grid_w and 0 <= ny < self.grid_h
   460→   460→            free = self._free_mask(ep.net_index) if in_bounds else None
   461→   461→            if not in_bounds or not bool(free[ep.layer, ny, nx]):
   462→   462→                reward += shaper.collision()
   463→   463→            else:
   464→   464→                ep.x, ep.y = nx, ny
   465→   465→                cell = (ep.x, ep.y, ep.layer)
   466→   466→                ep.visited.add(cell)
   467→   467→                ep.path.append(cell)
   468→   468→                new_dist = self._nearest_target_distance()
   469→   469→                reward += shaper.progress(prev_dist, new_dist)
   470→   470→                if cell in ep.targets:
   471→   471→                    ep.targets.remove(cell)
   472→   472→                    ep.done = True
   473→   473→                    ep.success = True
   474→   474→                    reward += shaper.success()
   475→   475→                    reward += shaper.terminal_length_cost(len(ep.path))
   476→   476→
   477→   477→        if not ep.done and ep.steps >= self.max_steps:
   478→   478→            ep.done = True
   479→   479→            ep.success = False
   480→   480→        truncated = ep.done and not ep.success
   481→   481→        return self._observation(), float(reward), ep.success, truncated, self._info()
   482→   482→
   483→   483→    def _info(self) -> dict:
   484→   484→        """Info dict accompanying :meth:`step`."""
   485→   485→        ep = self._episode
   486→   486→        dist_cells = self._nearest_target_distance()
   487→   487→        return {
   488→   488→            "net": self.net_names[ep.net_index] if 0 <= ep.net_index < len(self._nets) else "",
   489→   489→            "success": ep.success,
   490→   490→            "steps": ep.steps,
   491→   491→            "vias": ep.vias,
   492→   492→            "cells_visited": len(ep.visited),
   493→   493→            "path_len": len(ep.path),
   494→   494→            "dist_cells": dist_cells,
   495→   495→            "dist_mm": dist_cells * self.res,
   496→   496→            "position": (ep.x, ep.y, ep.layer),
   497→   497→            "done": ep.done,
   498→   498→        }

   502→   502→    def set_routes(self, routes: object) -> None:
   503→   503→        """Register already-routed tracks as obstacles.
   504→   504→
   505→   505→        Accepted inputs (normalised internally):
   506→   506→
   507→   507→        * a list of route entries;
   508→   508→        * a dict ``{"routes": [entry, ...]}``.
   509→   509→
   510→   510→        Entry formats:
   511→   511→
   512→   512→        * ``{"net": str, "layer": int, "points": [{"x": .., "y": ..}, ...]}``
   513→   513→        * ``{"net": str, "segments": [{"a": {"x","y","layer"}, "b": {...}},
   514→   514→          "vias": [{"x","y","from_layer","to_layer", ...}]}]``
   515→   515→          (the output format of :meth:`astar_route`).
   516→   516→
   517→   517→        Cells of a route never block the route's own net (a net can always be
   518→   518→        re-routed through / extended from its own previous geometry).
   519→   519→        """
   520→   520→        for entry in self._iter_route_entries(routes):
   521→   521→            name = str(entry.get("net", "") or "")
   522→   522→            if not name:
   523→   523→                continue
   524→   524→            cells = self._route_entry_cells(entry)
   525→   525→            bucket = self._routes_cells.setdefault(name, set())
   526→   526→            bucket.update(cells)
   527→   527→
   528→   528→    def clear_net_route(self, net_name: str) -> None:
   529→   529→        """Forget the registered route of one net (rip-up helper)."""
   530→   530→        self._routes_cells.pop(net_name, None)
   531→   531→        self._net_routes.pop(net_name, None)
   532→   532→
   533→   533→    def clear_routes(self) -> None:
   534→   534→        """Forget every registered route."""
   535→   535→        self._routes_cells.clear()
   536→   536→        self._net_routes.clear()

   538→   538→    @staticmethod
   539→   539→    def _iter_route_entries(routes: object) -> Iterator[dict]:
   540→   540→        """Yield normalised route entries from the accepted input shapes."""
   541→   541→        if isinstance(routes, dict):
   542→   542→            entries = routes.get("routes", routes if "net" in routes else [])
   543→   543→        else:
   544→   544→            entries = routes or []
   545→   545→        if not isinstance(entries, (list, tuple)):
   546→   546→            return
   547→   547→        for entry in entries:
   548→   548→            if isinstance(entry, dict):
   549→   549→                yield entry

   551→   551→    def _route_entry_cells(self, entry: dict) -> set[Cell]:
   552→   552→        """Extract blocked cells from one route entry."""
   553→   553→        cells: set[Cell] = set()
   554→   554→        segments = entry.get("segments")
   555→   555→        if isinstance(segments, (list, tuple)) and segments:
   556→   556→            for seg in segments:
   557→   557→                a = seg.get("a") or {}
   558→   558→                b = seg.get("b") or {}
   559→   559→                la = min(max(int(a.get("layer", 0) or 0), 0), self.layer_count - 1)
   560→   560→                lb = min(max(int(b.get("layer", la) or la), 0), self.layer_count - 1)
   561→   561→                ax, ay = self.to_cell(float(a.get("x", 0.0) or 0.0), float(a.get("y", 0.0) or 0.0))
   562→   562→                bx, by = self.to_cell(float(b.get("x", 0.0) or 0.0), float(b.get("y", 0.0) or 0.0))
   563→   563→                for (x, y) in self._raster_cells((ax, ay), (bx, by)):
   564→   564→                    cells.add((x, y, la))
   565→   565→                if lb != la:
   566→   566→                    cells.add((bx, by, la))
   567→   567→                    cells.add((bx, by, lb))
   568→   568→            for via in entry.get("vias") or []:
   569→   569→                vx, vy = self.to_cell(float(via.get("x", 0.0) or 0.0), float(via.get("y", 0.0) or 0.0))
   570→   570→                l1 = min(max(int(via.get("from_layer", 0) or 0), 0), self.layer_count - 1)
   571→   571→                l2 = min(max(int(via.get("to_layer", l1) or l1), 0), self.layer_count - 1)
   572→   572→                cells.add((vx, vy, l1))
   573→   573→                cells.add((vx, vy, l2))
   574→   574→            return cells
   575→   575→
   576→   576→        points = entry.get("points")
   577→   577→        if isinstance(points, (list, tuple)) and points:
   578→   578→            layer = min(max(int(entry.get("layer", 0) or 0), 0), self.layer_count - 1)
   579→   579→            cell_pts: list[tuple[int, int]] = [
   580→   580→                self.to_cell(float(p.get("x", 0.0) or 0.0), float(p.get("y", 0.0) or 0.0))
   581→   581→                for p in points
   582→   582→            ]
   583→   583→            for i in range(len(cell_pts) - 1):
   584→   584→                for (x, y) in self._raster_cells(cell_pts[i], cell_pts[i + 1]):
   585→   585→                    cells.add((x, y, layer))
   586→   586→            if len(cell_pts) == 1:
   587→   587→                cells.add((cell_pts[0][0], cell_pts[0][1], layer))
   588→   588→        return cells
   589→   589→
   590→   590→    @staticmethod
   591→   591→    def _raster_cells(c0: tuple[int, int], c1: tuple[int, int]) -> list[tuple[int, int]]:
   592→   592→        """Integer straight line between two cells (diagonal-safe raster)."""
   593→   593→        x0, y0 = c0
   594→   594→        x1, y1 = c1
   595→   595→        n = max(abs(x1 - x0), abs(y1 - y0))
   596→   596→        if n == 0:
   597→   597→            return [(x0, y0)]
   598→   598→        out: list[tuple[int, int]] = []
   599→   599→        for i in range(n + 1):
   600→   600→            t = i / n
   601→   601→            out.append((int(round(x0 + (x1 - x0) * t)), int(round(y0 + (y1 - y0) * t))))
   602→   602→        return out
   603→   603→
   604→   604→    # ---------------------------------------------------------------- A* router
   605→   605→
   606→   606→    def astar_route(self, net_index: int) -> dict:
   607→   607→        """Deterministically route one net with 4-connected multi-leg A*.
   608→   608→
   609→   609→        The net's pads are chained nearest-terminal-first: each leg runs an
   610→   610→        A* search (unit move cost, ``via_penalty`` per layer change, manhattan
   611→   611→        heuristic) from an already-connected pad to the nearest unconnected
   612→   612→        pad. Collinear runs are merged into segments.
   613→   613→
   614→   614→        Args:
   615→   615→            net_index: index of the net to route.
   616→   616→
   617→   617→        Returns:
   618→   618→            Route dict ``{"net", "segments", "vias", "length_mm",
   619→   619→            "completed"}`` where ``completed`` is True only when every pad
   620→   620→            was connected. Unreachable targets yield ``completed=False``
   621→   621→            with the (possibly empty) geometry of the legs that succeeded.
   622→   622→        """
   623→   623→        if not 0 <= net_index < len(self._nets):
   624→   624→            raise IndexError(f"net_index {net_index} out of range [0, {len(self._nets)})")
   625→   625→        net = self._nets[net_index]
   626→   626→        self.clear_net_route(net.name)
   627→   627→
   628→   628→        pad_cells: list[Cell] = []
   629→   629→        seen: set[Cell] = set()
   630→   630→        for pad in net.pads:
   631→   631→            cell = self._pad_cell(pad)
   632→   632→            if cell not in seen:
   633→   633→                seen.add(cell)
   634→   634→                pad_cells.append(cell)
   635→   635→
   636→   636→        empty = {"net": net.name, "segments": [], "vias": [], "length_mm": 0.0, "completed": False}
   637→   637→        if not pad_cells:
   638→   638→            return empty
   639→   639→        if len(pad_cells) == 1:
   640→   640→            route = {"net": net.name, "segments": [], "vias": [], "length_mm": 0.0, "completed": True}
   641→   641→            self._net_routes[net.name] = route
   642→   642→            return route
   643→   643→
   644→   644→        start = self._pick_start(pad_cells)
   645→   645→        connected: set[Cell] = {start}
   646→   646→        remaining: list[Cell] = [c for c in pad_cells if c != start]
   647→   647→        segments: list[dict] = []
   648→   648→        vias: list[dict] = []
   649→   649→        all_cells: set[Cell] = set()
   650→   650→        completed = True
   651→   651→
   652→   652→        while remaining:
   653→   653→            target = min(
   654→   654→                remaining,
   655→   655→                key=lambda t: (min(self._pad_dist(t, c) for c in connected), t),
   656→   656→            )
   657→   657→            source = min(
   658→   658→                connected,
   659→   659→                key=lambda c: (self._pad_dist(c, target), c),
   660→   660→            )
   661→   661→            leg = self._astar(net_index, source, target)
   662→   662→            if leg is None:
   663→   663→                completed = False
   664→   664→                break
   665→   665→            leg_cells = list(leg)
   666→   666→            leg_segments, leg_vias = self._cells_to_geometry(leg_cells, net.min_track_width_mm)
   667→   667→            segments.extend(leg_segments)
   668→   668→            vias.extend(leg_vias)
   669→   669→            bucket = self._routes_cells.setdefault(net.name, set())
   670→   670→            bucket.update(leg_cells)
   671→   671→            all_cells.update(leg_cells)
   672→   672→            connected.add(target)
   673→   673→            remaining.remove(target)
   674→   674→
   675→   675→        route = {
   676→   676→            "net": net.name,
   677→   677→            "segments": segments,
   678→   678→            "vias": vias,
   679→   679→            "length_mm": self._segments_length_mm(segments),
   680→   680→            "completed": completed,
   681→   681→        }
   682→   682→        self._net_routes[net.name] = route
   683→   683→        return route
   684→   684→
   685→   685→    def route_all(self, strategy: str = "astar") -> list[dict]:
   686→   686→        """Route every net, shortest first (progressive congestion).
   687→   687→
   688→   688→        Nets are ordered by their minimal pad-to-pad manhattan span (short
   689→   689→        nets first, the classic greedy heuristic) and every successful
   690→   690→        result is registered as an obstacle for the subsequent nets, which
   691→   691→        makes this the convenience entry point used by the gRPC service and
   692→   692→        the benchmark.
   693→   693→
   694→   694→        Args:
   695→   695→            strategy: only ``"astar"`` is currently supported.
   696→   696→
   697→   697→        Returns:
   698→   698→            The list of route dicts, in the order the nets were routed.
   699→   699→        """
   700→   700→        if strategy != "astar":
   701→   701→            raise ValueError(f"Unsupported routing strategy: {strategy!r} (expected 'astar')")
   702→   702→        order = sorted(
   703→   703→            range(len(self._nets)),
   704→   704→            key=lambda i: (self._net_span_cells(i), self._nets[i].name),
   705→   705→        )
   706→   706→        return [self.astar_route(i) for i in order]
   707→   707→
   708→   708→    def _net_span_cells(self, net_index: int) -> float:
   709→   709→        """Minimal pairwise manhattan span (cells) of a net's pads."""
   710→   710→        cells = [self._pad_cell(pad) for pad in self._nets[net_index].pads]
   711→   711→        if len(cells) < 2:
   712→   712→            return 0.0
   713→   713→        best = math.inf
   714→   714→        for i in range(len(cells)):
   715→   715→            for j in range(i + 1, len(cells)):
   716→   716→                best = min(best, self._pad_dist(cells[i], cells[j]))
   717→   717→        return best
   718→   718→
   719→   719→    def _astar(self, net_index: int, source: Cell, goal: Cell) -> list[Cell] | None:
   720→   720→        """Point-to-point A* on the free-cell grid (4-connected + vias).
   721→   721→
   722→   722→        Costs: 1 per orthogonal move, ``via_penalty`` per layer change.
   723→   723→        Heuristic: manhattan distance in x/y (admissible).
   724→   724→
   725→   725→        Returns:
   726→   726→            The list of cells from source to goal (inclusive), or ``None``
   727→   727→            when the goal is unreachable.
   728→   728→        """
   729→   729→        if source == goal:
   730→   730→            return [source]
   731→   731→        layers, height, width = self.layer_count, self.grid_h, self.grid_w
   732→   732→        free = self._free_mask(net_index)
   733→   733→        gx, gy, glayer = goal
   734→   734→        free[glayer, gy, gx] = True  # the goal pad is always routable
   735→   735→        via_cost = float(self.cfg.via_penalty)
   736→   736→
   737→   737→        def heuristic(cell: Cell) -> float:
   738→   738→            return abs(cell[0] - gx) + abs(cell[1] - gy)
   739→   739→
   740→   740→        counter = 0
   741→   741→        open_heap: list[tuple[float, float, int, Cell]] = [(heuristic(source), 0.0, counter, source)]
   742→   742→        g_score: dict[Cell, float] = {source: 0.0}
   743→   743→        parent: dict[Cell, Cell] = {}
   744→   744→        closed: set[Cell] = set()
   745→   745→
   746→   746→        while open_heap:
   747→   747→            _f, g, _c, cell = heapq.heappop(open_heap)
   748→   748→            if cell in closed:
   749→   749→                continue
   750→   750→            closed.add(cell)
   751→   751→            if cell == goal:
   752→   752→                path = [cell]
   753→   753→                while path[-1] != source:
   754→   754→                    path.append(parent[path[-1]])
   755→   755→                path.reverse()
   756→   756→                return path
   757→   757→            x, y, layer = cell
   758→   758→            neighbours: list[tuple[Cell, float]] = []
   759→   759→            for dx, dy in _MOVES_4:
   760→   760→                nx, ny = x + dx, y + dy
   761→   761→                if 0 <= nx < width and 0 <= ny < height and bool(free[layer, ny, nx]):
   762→   762→                    neighbours.append(((nx, ny, layer), 1.0))
   763→   763→            for dlayer in (1, -1):
   764→   764→                nl = layer + dlayer
   765→   765→                if 0 <= nl < layers and bool(free[nl, y, x]):
   766→   766→                    neighbours.append(((x, y, nl), via_cost))
   767→   767→            for neighbour, cost in neighbours:
   768→   768→                ng = g + cost
   769→   769→                if ng < g_score.get(neighbour, math.inf):
   770→   770→                    g_score[neighbour] = ng
   771→   771→                    parent[neighbour] = cell
   772→   772→                    counter += 1
   773→   773→                    heapq.heappush(
   774→   774→                        open_heap, (ng + heuristic(neighbour), ng, counter, neighbour)
   775→   775→                    )
   776→   776→        return None
   777→   777→

   779→   779→
   780→   780→    def path_to_route(
   781→   781→        self, net_index: int, cells: Sequence[Cell], register: bool = True
   782→   782→    ) -> dict:
   783→   783→        """Convert an episode cell path into the standard route dict.
   784→   784→
   785→   785→        Used to export RL rollouts. ``completed`` is True only when the path
   786→   786→        covers every distinct pad cell of the net (single-target rollouts of
   787→   787→        nets with more than two pads are reported as partial). When
   788→   788→        ``register`` is True the geometry is registered as an obstacle for
   789→   789→        the subsequent nets, mirroring :meth:`astar_route`.
   790→   790→        """
   791→   791→        net = self._nets[net_index]
   792→   792→        path = list(cells)
   793→   793→        if not path:
   794→   794→            return {"net": net.name, "segments": [], "vias": [], "length_mm": 0.0, "completed": False}
   795→   795→        segments, vias = self._cells_to_geometry(path, net.min_track_width_mm)
   796→   796→        pad_cells = {self._pad_cell(pad) for pad in net.pads}
   797→   797→        completed = pad_cells.issubset(set(path)) if pad_cells else False
   798→   798→        route = {
   799→   799→            "net": net.name,
   800→   800→            "segments": segments,
   801→   801→            "vias": vias,
   802→   802→            "length_mm": self._segments_length_mm(segments),
   803→   803→            "completed": completed,
   804→   804→        }
   805→   805→        if register:
   806→   806→            bucket = self._routes_cells.setdefault(net.name, set())
   807→   807→            bucket.update(path)
   808→   808→            self._net_routes[net.name] = route
   809→   809→        return route
   810→   810→
   811→   811→    def _cells_to_geometry(
   812→   812→        self, cells: Sequence[Cell], width_mm: float
   813→   813→    ) -> tuple[list[dict], list[dict]]:
   814→   814→        """Merge a cell path into collinear segments + via descriptors."""
   815→   815→        segments: list[dict] = []
   816→   816→        vias: list[dict] = []
   817→   817→        if not cells:
   818→   818→            return segments, vias
   819→   819→
   820→   820→        def emit(a: Cell, b: Cell) -> None:
   821→   821→            if a == b:
   822→   822→                return
   823→   823→            ax_mm, ay_mm = self.to_mm(a[0], a[1])
   824→   824→            bx_mm, by_mm = self.to_mm(b[0], b[1])
   825→   825→            segments.append(
   826→   826→                {
   827→   827→                    "a": {"x": ax_mm, "y": ay_mm, "layer": a[2]},
   828→   828→                    "b": {"x": bx_mm, "y": by_mm, "layer": b[2]},
   829→   829→                    "width_mm": float(width_mm),
   830→   830→                }
   831→   831→            )
   832→   832→
   833→   833→        seg_start = cells[0]
   834→   834→        prev = cells[0]
   835→   835→        direction: tuple[int, int] | None = None
   836→   836→        for cell in cells[1:]:
   837→   837→            if cell[0] == prev[0] and cell[1] == prev[1] and cell[2] != prev[2]:
   838→   838→                # Layer change on the spot: close the segment, emit the via.
   839→   839→                emit(seg_start, prev)
   840→   840→                seg_start = cell
   841→   841→                direction = None
   842→   842→                x_mm, y_mm = self.to_mm(cell[0], cell[1])
   843→   843→                vias.append(
   844→   844→                    {
   845→   845→                        "x": x_mm,
   846→   846→                        "y": y_mm,
   847→   847→                        "from_layer": prev[2],
   848→   848→                        "to_layer": cell[2],
   849→   849→                        "diameter_mm": round(max(2.0 * width_mm, 0.6), 4),
   850→   850→                        "drill_mm": round(max(width_mm, 0.3), 4),
   851→   851→                    }
   852→   852→                )
   853→   853→            else:
   854→   854→                step = (cell[0] - prev[0], cell[1] - prev[1])
   855→   855→                if direction is not None and step != direction:
   856→   856→                    emit(seg_start, prev)
   857→   857→                    seg_start = prev
   858→   858→                direction = step
   859→   859→            prev = cell
   860→   860→        emit(seg_start, prev)
   861→   861→        return segments, vias
   862→   862→
   863→   863→    @staticmethod
   864→   864→    def _segments_length_mm(segments: Sequence[dict]) -> float:
   865→   865→        """Total Euclidean length of a segment list (mm)."""
   866→   866→        total = 0.0
   867→   867→        for seg in segments:
   868→   868→            a, b = seg.get("a") or {}, seg.get("b") or {}
   869→   869→            total += math.hypot(
   870→   870→                float(b.get("x", 0.0)) - float(a.get("x", 0.0)),
   871→   871→                float(b.get("y", 0.0)) - float(a.get("y", 0.0)),
   872→   872→            )
   873→   873→        return round(total, 4)
   874→   874→

   877→   877→    def render(self, layer: int = 0) -> str:
   878→   878→        """ASCII rendering of one layer (debug helper).
   879→   879→
   880→   880→        Legend: ``S`` start, ``T`` target, ``*`` agent, ``#`` obstacle,
   881→   881→        ``o`` foreign pad halo, ``.`` free cell.
   882→   882→        """
   883→   883→        ep = self._episode
   884→   884→        free = self._free_mask(ep.net_index) if 0 <= ep.net_index < len(self._nets) else None
   885→   885→        rows: list[str] = []
   886→   886→        for y in range(self.grid_h):
   887→   887→            row_chars: list[str] = []
   888→   888→            for x in range(self.grid_w):
   889→   889→                cell = (x, y, layer)
   890→   890→                if ep.start == cell:
   891→   891→                    row_chars.append("S")
   892→   892→                elif any(t[:2] == (x, y) for t in ep.targets):
   893→   893→                    row_chars.append("T")
   894→   894→                elif (ep.x, ep.y, ep.layer) == cell:
   895→   895→                    row_chars.append("*")
   896→   896→                elif free is not None and not bool(free[layer, y, x]):
   897→   897→                    row_chars.append("#")
   898→   898→                else:
   899→   899→                    row_chars.append(".")
   900→   900→            rows.append("".join(row_chars))
   901→   901→        return "\n".join(rows)
