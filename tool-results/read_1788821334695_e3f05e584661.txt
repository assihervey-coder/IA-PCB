    18→
    19→// sNode is one node of the s-expression tree. A list node keeps its direct
    20→// atoms (the first one being the tag) and its direct sub-lists; a bare atom
    21→// is not represented outside of the atoms slice of its parent.
    22→type sNode struct {
    23→        atoms    []string
    24→        children []*sNode
    25→}
    26→
    27→// tag returns the tag of a list node ("segment", "pad", ...) or "".
    28→func (n *sNode) tag() string {
    29→        if n == nil || len(n.atoms) == 0 {
    30→                return ""
    31→        }
    32→        return n.atoms[0]
    33→}
    34→
    35→// args returns the atoms following the tag.
    36→func (n *sNode) args() []string {
    37→        if n == nil || len(n.atoms) <= 1 {
    38→                return nil
    39→        }
    40→        return n.atoms[1:]
    41→}
    42→
    43→// arg returns the i-th atom after the tag, or "" when missing.
    44→func (n *sNode) arg(i int) string {
    45→        args := n.args()
    46→        if i < 0 || i >= len(args) {
    47→                return ""
    48→        }
    49→        return args[i]
    50→}
    51→
    52→// child returns the first direct sub-list carrying the given tag.
    53→func (n *sNode) child(name string) *sNode {
    54→        for _, c := range n.children {
    55→                if c.tag() == name {
    56→                        return c
    57→                }
    58→        }
    59→        return nil
    60→}
    61→
    62→// childrenWithTag returns every direct sub-list carrying the given tag.
    63→func (n *sNode) childrenWithTag(name string) []*sNode {
    64→        var out []*sNode
    65→        for _, c := range n.children {
    66→                if c.tag() == name {
    67→                        out = append(out, c)
    68→                }
    69→        }
    70→        return out
    71→}
    72→
    73→// hasAtomIn reports whether one of the direct atoms after the tag equals v.
    74→func (n *sNode) hasAtomIn(v string) bool {
    75→        for _, a := range n.args() {
    76→                if a == v {
    77→                        return true
    78→                }
    79→        }
    80→        return false
    81→}

    83→// tokenizeSexpr splits an s-expression payload into "(", ")" and atom
    84→// tokens (quoted strings are unescaped).
    85→func tokenizeSexpr(data []byte) ([]string, error) {
    86→        var toks []string
    87→        i := 0
    88→        for i < len(data) {
    89→                c := data[i]
    90→                switch {
    91→                case c == '(' || c == ')':
    92→                        toks = append(toks, string(c))
    93→                        i++
    94→                case c == '"':
    95→                        s, next, err := readQuoted(data, i)
    96→                        if err != nil {
    97→                                return nil, err
    98→                        }
    99→                        toks = append(toks, s)
   100→                        i = next
   101→                case c == ' ' || c == '\t' || c == '\n' || c == '\r':
   102→                        i++
   103→                default:
   104→                        j := i
   105→                        for j < len(data) && data[j] != '(' && data[j] != ')' &&
   106→                                data[j] != '"' && data[j] != ' ' && data[j] != '\t' &&
   107→                                data[j] != '\n' && data[j] != '\r' {
   108→                                j++
   109→                        }
   110→                        toks = append(toks, string(data[i:j]))
   111→                        i = j
   112→                }
   113→        }
   114→        return toks, nil
   115→}
   116→
   117→// readQuoted reads a double-quoted atom starting at position start,
   118→// resolving the standard backslash escapes.
   119→func readQuoted(data []byte, start int) (string, int, error) {
   120→        var sb strings.Builder
   121→        i := start + 1
   122→        for i < len(data) {
   123→                c := data[i]
   124→                switch c {
   125→                case '"':
   126→                        return sb.String(), i + 1, nil
   127→                case '\\':
   128→                        if i+1 >= len(data) {
   129→                                return "", 0, fmt.Errorf("s-expression : échappement tronqué")
   130→                        }
   131→                        switch data[i+1] {
   132→                        case 'n':
   133→                                sb.WriteByte('\n')
   134→                        case 't':
   135→                                sb.WriteByte('\t')
   136→                        case 'r':
   137→                                sb.WriteByte('\r')
   138→                        default:
   139→                                sb.WriteByte(data[i+1])
   140→                        }
   141→                        i += 2
   142→                default:
   143→                        sb.WriteByte(c)
   144→                        i++
   145→                }
   146→        }
   147→        return "", 0, fmt.Errorf("s-expression : chaîne non terminée")
   148→}
   149→
   150→// parseSexpr parses a payload into a virtual root node holding every
   151→// top-level list (and every loose atom) as children.
   152→func parseSexpr(data []byte) (*sNode, error) {
   153→        toks, err := tokenizeSexpr(data)
   154→        if err != nil {
   155→                return nil, err
   156→        }
   157→        root := &sNode{}
   158→        pos := 0
   159→        for pos < len(toks) {
   160→                node, next, err := parseNode(toks, pos)
   161→                if err != nil {
   162→                        return nil, err
   163→                }
   164→                if node != nil {
   165→                        root.children = append(root.children, node)
   166→                }
   167→                pos = next
   168→        }
   169→        if len(root.children) == 0 {
   170→                return nil, fmt.Errorf("s-expression : fichier vide")
   171→        }
   172→        return root, nil
   173→}
   174→
   175→// parseNode parses one node starting at pos and returns the next position.
   176→func parseNode(toks []string, pos int) (*sNode, int, error) {
   177→        if pos >= len(toks) {
   178→                return nil, pos, fmt.Errorf("s-expression : fin inattendue")
   179→        }
   180→        if toks[pos] == ")" {
   181→                return nil, pos + 1, fmt.Errorf("s-expression : parenthèse fermante inattendue")
   182→        }
   183→        if toks[pos] != "(" {
   184→                // Atome isolé : représenté comme une liste dégénérée d'un seul atome,
   185→                // ignorée par les lecteurs (les atomes sont portés par leur parent).
   186→                return &sNode{atoms: []string{toks[pos]}}, pos + 1, nil
   187→        }
   188→        pos++
   189→        node := &sNode{}
   190→        for pos < len(toks) {
   191→                switch toks[pos] {
   192→                case ")":
   193→                        return node, pos + 1, nil
   194→                case "(":
   195→                        child, next, err := parseNode(toks, pos)
   196→                        if err != nil {
   197→                                return nil, pos, err
   198→                        }
   199→                        node.children = append(node.children, child)
   200→                        pos = next
   201→                default:
   202→                        node.atoms = append(node.atoms, toks[pos])
   203→                        pos++
   204→                }
   205→        }
   206→        return nil, pos, fmt.Errorf("s-expression : parenthèses déséquilibrées")
   207→}

   209→// sexprFloat parses a numeric atom with a zero fallback.
   210→func sexprFloat(s string) float64 {
   211→        v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
   212→        if err != nil || math.IsNaN(v) || math.IsInf(v, 0) {
   213→                return 0
   214→        }
   215→        return v
   216→}
   217→
   218→// sexprInt parses an integer atom with a zero fallback.
   219→func sexprInt(s string) int {
   220→        v, err := strconv.Atoi(strings.TrimSpace(s))
   221→        if err != nil {
   222→                return 0
   223→        }
   224→        return v
   225→}

   231→// kicadLayerIndex maps a KiCad layer name onto a copper layer index
   232→// (0 = F.Cu, layerCount-1 = B.Cu, InN.Cu = N).
   233→func kicadLayerIndex(name string, layerCount int) int {
   234→        switch {
   235→        case name == "F.Cu" || name == "Top":
   236→                return 0
   237→        case name == "B.Cu" || name == "Bottom":
   238→                if layerCount <= 1 {
   239→                        return 0
   240→                }
   241→                return layerCount - 1
   242→        case strings.HasPrefix(name, "In") && strings.HasSuffix(name, ".Cu"):
   243→                if n, err := strconv.Atoi(strings.TrimSuffix(strings.TrimPrefix(name, "In"), ".Cu")); err == nil {
   244→                        if n > 0 && n < layerCount-1 {
   245→                                return n
   246→                        }
   247→                }
   248→                return 0
   249→        default:
   250→                return 0
   251→        }
   252→}
   253→
   254→// padShapeFromKicad maps a KiCad pad shape onto the domain enum.
   255→func padShapeFromKicad(shape string) string {
   256→        switch strings.ToLower(strings.TrimSpace(shape)) {
   257→        case "circle":
   258→                return "circle"
   259→        case "oval", "ellipse":
   260→                return "oval"
   261→        case "roundrect", "round_rect":
   262→                return "roundrect"
   263→        case "rect":
   264→                return "rect"
   265→        default:
   266→                return "rect"
   267→        }
   268→}

   270→// stripLibraryPrefix removes the "Lib:" prefix of a footprint name.
   271→func stripLibraryPrefix(name string) string {
   272→        if idx := strings.LastIndex(name, ":"); idx >= 0 && idx < len(name)-1 {
   273→                return name[idx+1:]
   274→        }
   275→        return name
   276→}

   308→// geometryAABB is a minimal bounding box used for the board outline.
   309→type geometryAABB struct {

   313→func (b *geometryAABB) unionPoint(x, y float64) {
   314→        b.MinX = math.Min(b.MinX, x)
   315→        b.MinY = math.Min(b.MinY, y)
   316→        b.MaxX = math.Max(b.MaxX, x)
   317→        b.MaxY = math.Max(b.MaxY, y)
   318→}
   319→
   320→func (b *geometryAABB) union(o geometryAABB) {
   321→        b.MinX = math.Min(b.MinX, o.MinX)
   322→        b.MinY = math.Min(b.MinY, o.MinY)
   323→        b.MaxX = math.Max(b.MaxX, o.MaxX)
   324→        b.MaxY = math.Max(b.MaxY, o.MaxY)
   325→}

   337→// readKiCadPCB parses a .kicad_pcb file (contract §7 subset) into the
   338→// schematic + board pair.
   339→func readKiCadPCB(data []byte, path string) (schematicapp.ImportResult, error) {
   340→        root, err := parseSexpr(data)
   341→        if err != nil {
   342→                return schematicapp.ImportResult{}, fmt.Errorf("kicad %s : %w", path, err)
   343→        }
   344→        pcb := root.child("kicad_pcb")
   345→        if pcb == nil {
   346→                return schematicapp.ImportResult{}, fmt.Errorf(
   347→                        "kicad %s : racine (kicad_pcb ...) introuvable", path)
   348→        }
   349→
   350→        st := &kicadParseState{
   351→                layerCount: 2,
   352→                netNames:   map[int]string{},
   353→                netUsed:    map[int]bool{},
   354→                netIDs:     map[string]int{},
   355→                nextNetID:  netSyntheticBase,
   356→                conns:      map[int][]schematicPinRef{},
   357→        }
   358→
   359→        // Nombre de couches cuivre : entrées "xxx.Cu" du bloc (layers ...).
   360→        if layers := pcb.child("layers"); layers != nil {
   361→                cu := 0
   362→                for _, l := range layers.children {
   363→                        name := l.arg(0)
   364→                        if strings.HasSuffix(name, ".Cu") {
   365→                                cu++
   366→                        }
   367→                }
   368→                if cu >= 1 && cu <= 34 {
   369→                        st.layerCount = cu
   370→                }
   371→        }
   372→
   373→        // Déclarations : (net N "NAME") — KiCad <= 9 — ou (net "NAME") —
   374→        // KiCad 10 (version 20260206), qui ne déclare plus les nets au niveau
   375→        // racine ; ils sont alors synthétisés depuis les références cuivre.
   376→        for _, n := range pcb.childrenWithTag("net") {
   377→                if num, name, ok := st.resolveNetRef(n); ok && num > 0 && name != "" {
   378→                        st.registerNet(num, name)
   379→                }
   380→        }
   381→
   382→        for _, fp := range pcb.childrenWithTag("footprint") {
   383→                st.readFootprint(fp)
   384→        }
   385→        for _, seg := range pcb.childrenWithTag("segment") {
   386→                st.readSegment(seg)
   387→        }
   388→        for _, via := range pcb.childrenWithTag("via") {
   389→                st.readVia(via)
   390→        }
   391→
   392→        width, height := st.boardOutline(pcb)
   393→        board, err := domainlayout.NewBoard(width, height, st.layerCount)
   394→        if err != nil {
   395→                return schematicapp.ImportResult{}, fmt.Errorf("kicad %s : %w", path, err)
   396→        }
   397→
   398→        // Empreintes -> composants placés.
   399→        for i := range st.footprints {
   400→                kf := &st.footprints[i]
   401→                bodyW, bodyH := kicadBodySize(kf.pads)
   402→                pl := domainlayout.PlacedComponent{
   403→                        Ref: kf.ref,
   404→                        Footprint: domainlayout.Footprint{
   405→                                Name:         kf.name,
   406→                                Value:        kf.value,
   407→                                Pads:         kf.pads,
   408→                                BodyWidthMM:  bodyW,
   409→                                BodyHeightMM: bodyH,
   410→                                HeightMM:     domainlayout.DefaultBoardHeightMM,
   411→                        },
   412→                        X:        kf.x,
   413→                        Y:        kf.y,
   414→                        Rotation: kf.rot,
   415→                }
   416→                if err := board.AddComponent(pl); err != nil {
   417→                        st.warnings = append(st.warnings, fmt.Sprintf("empreinte ignorée : %v", err))
   418→                        continue
   419→                }
   420→        }
   421→
   422→        // Pistes et vias — fusion cosmétique des segments contigus/colinéaires
   423→        // en pistes multi-points (géométrie inchangée, jonctions préservées).
   424→        st.tracks = mergeTracks(st.tracks)
   425→        for _, t := range st.tracks {
   426→                board.AddTrack(t)
   427→        }
   428→        for _, v := range st.vias {
   429→                if err := board.AddVia(v); err != nil {
   430→                        st.warnings = append(st.warnings, fmt.Sprintf("via ignoré : %v", err))
   431→                }
   432→        }
   433→
   434→        // Schéma : composants + nets reconstitués depuis les pastilles.
   435→        sch := domainschematic.New(schematicBaseName(path, "kicad"))
   436→        for i := range st.footprints {
   437→                kf := &st.footprints[i]
   438→                if err := sch.AddComponent(domainschematic.Component{
   439→                        Ref:       kf.ref,
   440→                        Value:     kf.value,
   441→                        Footprint: kf.name,
   442→                        X:         kf.x,
   443→                        Y:         kf.y,
   444→                        Rotation:  kf.rot,
   445→                }); err != nil {
   446→                        st.warnings = append(st.warnings, fmt.Sprintf("composant ignoré : %v", err))
   447→                }
   448→        }
   449→        for _, num := range st.netOrder {
   450→                if !st.netUsed[num] {
   451→                        continue
   452→                }
   453→                name := st.netNames[num]
   454→                if name == "" {
   455→                        name = fmt.Sprintf("NET_%d", num)
   456→                }
   457→                net := domainschematic.Net{Name: name, Class: domainschematic.ClassDefault}
   458→                for _, conn := range st.conns[num] {
   459→                        net.Connections = append(net.Connections, domainschematic.PinRef{
   460→                                ComponentRef: conn.ref, PinNumber: conn.pin,
   461→                        })
   462→                }
   463→                if err := sch.AddNet(net); err != nil {
   464→                        st.warnings = append(st.warnings, fmt.Sprintf("net ignoré : %v", err))
   465→                }
   466→        }
   467→
   468→        res := schematicapp.ImportResult{
   469→                Format:      "kicad",
   470→                Schematic:   sch,
   471→                Board:       board,
   472→                Constraints: domainconstraints.NewDefault(),
   473→                Warnings:    st.warnings,
   474→        }
   475→        return res, nil
   476→}

   478→// readFootprint parses one (footprint ...) node.
   479→func (st *kicadParseState) readFootprint(fp *sNode) {
   480→        kf := kicadFootprint{
   481→                name:  stripLibraryPrefix(fp.arg(0)),
   482→                front: true,
   483→        }
   484→
   485→        if layer := fp.child("layer"); layer != nil {
   486→                kf.front = layer.arg(0) != "B.Cu"
   487→        }
   488→        if at := fp.child("at"); at != nil {
   489→                kf.x = sexprFloat(at.arg(0))
   490→                kf.y = sexprFloat(at.arg(1))
   491→                if at.arg(2) != "" {
   492→                        kf.rot = sexprFloat(at.arg(2))
   493→                }
   494→        }
   495→
   496→        ref := fmt.Sprintf("FP%d", len(st.footprints)+1)
   497→        value := ""
   498→        // pcbnew ≥ 7 stocke référence et valeur dans des (property "Reference" /
   499→        // "Value" …) ; pcbnew ≤ 6 utilisait (fp_text reference/value …).
   500→        for _, prop := range fp.childrenWithTag("property") {
   501→                switch prop.arg(0) {
   502→                case "Reference":
   503→                        if v := prop.arg(1); v != "" {
   504→                                ref = v
   505→                        }
   506→                case "Value":
   507→                        if v := prop.arg(1); v != "" {
   508→                                value = v
   509→                        }
   510→                }
   511→        }
   512→        for _, txt := range fp.childrenWithTag("fp_text") {
   513→                switch txt.arg(0) {
   514→                case "reference":
   515→                        if v := txt.arg(1); v != "" {
   516→                                ref = v
   517→                        }
   518→                case "value":
   519→                        value = txt.arg(1)
   520→                }
   521→        }
   522→        kf.ref, kf.value = ref, value
   523→
   524→        lastLayer := st.layerCount - 1
   525→        for _, padNode := range fp.childrenWithTag("pad") {
   526→                args := padNode.args()
   527→                padName := ""
   528→                padType := "smd"
   529→                padShape := "rect"
   530→                if len(args) > 0 {
   531→                        padName = args[0]
   532→                }
   533→                if len(args) > 1 {
   534→                        padType = args[1]
   535→                }
   536→                if len(args) > 2 {
   537→                        padShape = args[2]
   538→                }
   539→
   540→                pad := domainlayout.Pad{
   541→                        Name:  padName,
   542→                        Shape: domainlayout.PadShape(padShapeFromKicad(padShape)),
   543→                }
   544→                if at := padNode.child("at"); at != nil {
   545→                        pad.X = sexprFloat(at.arg(0))
   546→                        pad.Y = sexprFloat(at.arg(1))
   547→                        pad.Rotation = sexprFloat(at.arg(2))
   548→                }
   549→                if size := padNode.child("size"); size != nil {
   550→                        pad.Width = sexprFloat(size.arg(0))
   551→                        pad.Height = sexprFloat(size.arg(1))
   552→                }
   553→                if drill := padNode.child("drill"); drill != nil {
   554→                        // (drill 1.0) ou (drill oval 1.0 0.6)
   555→                        d := sexprFloat(drill.arg(0))
   556→                        if d == 0 {
   557→                                d = sexprFloat(drill.arg(1))
   558→                        }
   559→                        _ = d // le perçage est porté par le diamètre côté domaine
   560→                }
   561→
   562→                layersNode := padNode.child("layers")
   563→                thru := padType == "thru_hole" || padType == "np_thru_hole"
   564→                if layersNode != nil && layersNode.hasAtomIn("*.Cu") {
   565→                        thru = true
   566→                }
   567→                switch {
   568→                case thru:
   569→                        pad.Layer = -1
   570→                case !kf.front:
   571→                        pad.Layer = lastLayer
   572→                default:
   573→                        pad.Layer = 0
   574→                }
   575→
   576→                // (net N "NAME") — KiCad <= 9 — ou (net "NAME") — KiCad 10 :
   577→                // association électrique de la pastille.
   578→                if netNode := padNode.child("net"); netNode != nil {
   579→                        if num, name, ok := st.resolveNetRef(netNode); ok && num > 0 &&
   580→                                (name != "" || st.netNames[num] != "") {
   581→                                st.registerNet(num, name)
   582→                                pad.Net = st.netName(num)
   583→                                st.netUsed[num] = true
   584→                                st.conns[num] = append(st.conns[num],
   585→                                        schematicPinRef{ref: kf.ref, pin: padName})
   586→                        }
   587→                }
   588→                kf.pads = append(kf.pads, pad)
   589→        }
   590→
   591→        st.footprints = append(st.footprints, kf)
   592→}

   594→// netName resolves the name of a net number, with a deterministic fallback.
   595→func (st *kicadParseState) netName(num int) string {
   596→        if name := st.netNames[num]; name != "" {
   597→                return name
   598→        }
   599→        return fmt.Sprintf("NET_%d", num)
   600→}

   602→// resolveNetRef interprets a (net ...) node in the two supported formats :
   603→//   - KiCad <= 9 : (net N "NAME") — numéro explicite, nom optionnel ;
   604→//   - KiCad >= 10 (version 20260206) : (net "NAME") — nom sans numéro.
   605→//
   606→// Pour le format par nom seul, un numéro synthétique stable (>=
   607→// netSyntheticBase) est attribué à la première rencontre, si bien que la
   608→// suite du lecteur (netOrder, netUsed, conns) continue de travailler sur
   609→// des entiers. ok=false pour un nœud vide ou un nom vide.
   610→func (st *kicadParseState) resolveNetRef(node *sNode) (int, string, bool) {
   611→        args := node.args()
   612→        if len(args) == 0 {
   613→                return 0, "", false
   614→        }
   615→        if num, err := strconv.Atoi(args[0]); err == nil {
   616→                name := ""
   617→                if len(args) > 1 {
   618→                        name = args[1]
   619→                }
   620→                return num, name, true
   621→        }
   622→        name := args[0]
   623→        if name == "" {
   624→                return 0, "", false
   625→        }
   626→        num, seen := st.netIDs[name]
   627→        if !seen {
   628→                st.nextNetID++
   629→                num = st.nextNetID
   630→                st.netIDs[name] = num
   631→        }
   632→        return num, name, true
   633→}

   635→// registerNet enregistre une paire (numéro, nom) : netOrder à la première
   636→// rencontre, netNames sans effacer un nom déjà connu.
   637→func (st *kicadParseState) registerNet(num int, name string) {
   638→        if num <= 0 {
   639→                return
   640→        }
   641→        if _, seen := st.netNames[num]; !seen {
   642→                st.netOrder = append(st.netOrder, num)
   643→        }
   644→        if name != "" {
   645→                st.netNames[num] = name
   646→        }
   647→}

   649→// readSegment parses one (segment (start) (end) (width) (layer) (net)) node.
   650→func (st *kicadParseState) readSegment(seg *sNode) {
   651→        start := seg.child("start")
   652→        end := seg.child("end")
   653→        if start == nil || end == nil {
   654→                return
   655→        }
   656→        track := domainlayout.Track{
   657→                Width: 0.25,
   658→                Layer: 0,
   659→                Points: []domainlayout.TrackPoint{
   660→                        {X: sexprFloat(start.arg(0)), Y: sexprFloat(start.arg(1))},
   661→                        {X: sexprFloat(end.arg(0)), Y: sexprFloat(end.arg(1))},
   662→                },
   663→        }
   664→        if w := seg.child("width"); w != nil {
   665→                track.Width = sexprFloat(w.arg(0))
   666→        }
   667→        if layer := seg.child("layer"); layer != nil {
   668→                track.Layer = kicadLayerIndex(layer.arg(0), st.layerCount)
   669→        }
   670→        if net := seg.child("net"); net != nil {
   671→                if num, _, ok := st.resolveNetRef(net); ok && num > 0 {
   672→                        track.Net = st.netName(num)
   673→                        st.netUsed[num] = true
   674→                }
   675→        }
   676→        st.tracks = append(st.tracks, track)
   677→}

   679→// readVia parses one (via (at) (size) (drill) (layers) (net)) node.
   680→func (st *kicadParseState) readVia(via *sNode) {
   681→        at := via.child("at")
   682→        if at == nil {
   683→                return
   684→        }
   685→        diameter := sexprFloat(via.arg(0)) // forme via micro/blind : diamètre direct
   686→        if size := via.child("size"); size != nil {
   687→                diameter = sexprFloat(size.arg(0))
   688→        }
   689→        drill := 0.0
   690→        if d := via.child("drill"); d != nil {
   691→                drill = sexprFloat(d.arg(0))
   692→        }
   693→        if drill <= 0 {
   694→                drill = diameter / 2
   695→        }
   696→        if diameter <= drill {
   697→                diameter = drill + 0.2
   698→        }
   699→
   700→        fromLayer, toLayer := 0, st.layerCount-1
   701→        if layers := via.child("layers"); layers != nil && len(layers.args()) >= 2 {
   702→                fromLayer = kicadLayerIndex(layers.arg(0), st.layerCount)
   703→                toLayer = kicadLayerIndex(layers.arg(1), st.layerCount)
   704→                if fromLayer == toLayer {
   705→                        toLayer = st.layerCount - 1
   706→                }
   707→        }
   708→
   709→        v := domainlayout.Via{
   710→                X:         sexprFloat(at.arg(0)),
   711→                Y:         sexprFloat(at.arg(1)),
   712→                FromLayer: fromLayer,
   713→                ToLayer:   toLayer,
   714→                Diameter:  diameter,
   715→                Drill:     drill,
   716→        }
   717→        if net := via.child("net"); net != nil {
   718→                if num, _, ok := st.resolveNetRef(net); ok && num > 0 {
   719→                        v.Net = st.netName(num)
   720→                        st.netUsed[num] = true
   721→                }
   722→        }
   723→        st.vias = append(st.vias, v)
   724→}

   726→// boardOutline derives the board size from the Edge.Cuts rectangles,
   727→// defaulting to 100 x 80 mm.
   728→func (st *kicadParseState) boardOutline(pcb *sNode) (float64, float64) {
   729→        for _, rect := range pcb.childrenWithTag("gr_rect") {
   730→                layer := rect.child("layer")
   731→                if layer != nil && layer.arg(0) != "Edge.Cuts" {
   732→                        continue
   733→                }
   734→                start, end := rect.child("start"), rect.child("end")
   735→                if start == nil || end == nil {
   736→                        continue
   737→                }
   738→                box := geometryAABB{
   739→                        MinX: math.Min(sexprFloat(start.arg(0)), sexprFloat(end.arg(0))),
   740→                        MinY: math.Min(sexprFloat(start.arg(1)), sexprFloat(end.arg(1))),
   741→                        MaxX: math.Max(sexprFloat(start.arg(0)), sexprFloat(end.arg(0))),
   742→                        MaxY: math.Max(sexprFloat(start.arg(1)), sexprFloat(end.arg(1))),
   743→                }
   744→                if st.outline == nil {
   745→                        st.outline = &geometryAABB{}
   746→                        *st.outline = box
   747→                } else {
   748→                        st.outline.union(box)
   749→                }
   750→        }
   751→        if st.outline == nil {
   752→                return 100, 80
   753→        }
   754→        w := st.outline.MaxX - st.outline.MinX
   755→        h := st.outline.MaxY - st.outline.MinY
   756→        if w <= 0 || h <= 0 {
   757→                return 100, 80
   758→        }
   759→        return w, h
   760→}

   762→// kicadBodySize derives the footprint body size from the pad bounding box,
   763→// falling back to 2 x 2 mm.
   764→func kicadBodySize(pads []domainlayout.Pad) (float64, float64) {
   765→        minX, minY, maxX, maxY := 0.0, 0.0, 0.0, 0.0
   766→        for i, p := range pads {
   767→                x0, x1 := p.X-p.Width/2, p.X+p.Width/2
   768→                y0, y1 := p.Y-p.Height/2, p.Y+p.Height/2
   769→                if i == 0 {
   770→                        minX, maxX, minY, maxY = x0, x1, y0, y1
   771→                        continue
   772→                }
   773→                minX = math.Min(minX, x0)
   774→                minY = math.Min(minY, y0)
   775→                maxX = math.Max(maxX, x1)
   776→                maxY = math.Max(maxY, y1)
   777→        }
   778→        if len(pads) == 0 || maxX-minX <= 0 || maxY-minY <= 0 {
   779→                return 2, 2
   780→        }
   781→        return maxX - minX, maxY - minY
   782→}
